// 0.25 F1 PR 2 (#943) fixture: an out declaration in a call that is a statement by itself (W1
// finding 1). Its bare binding took the next call as its initializer; since PR 2 the binding
// starts at `default`, so the lowering is native, and so is a following array declaration
// (Codex PR 2 round 2).
public static class Probe
{
    static string Log = "";

    static void Use(int x) { Log += x + ";"; }

    public static string Statement(string s)
    {
        int.TryParse(s, out var n);
        Use(n);
        int.TryParse("3", out int m);
        Use(m);
        return Log;
    }

    static int Set(out int[] n) { n = new[] { 7 }; return 1; }

    public static string ArrayDeclaration()
    {
        int[] a = new[] { Set(out var n) };
        return n[0] + ":" + a[0];
    }

    public static string Run()
    {
        return Statement("5") + "|" + Statement("x") + "|" + ArrayDeclaration();
    }
}
