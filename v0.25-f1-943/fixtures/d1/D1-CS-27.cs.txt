public static class Probe
{
    static string log = "";
    static string Note(string s) { log += s; return s; }
    static string Read(in object slot, string tag) { return tag + "=" + slot; }
    static string ReadRef(ref object slot, string tag) { return tag + "=" + slot; }
    public static string Run()
    {
        object[] a = new string[] { "x" };
        return Read(in a[0], Note("n")) + "|" + log;
    }
}
