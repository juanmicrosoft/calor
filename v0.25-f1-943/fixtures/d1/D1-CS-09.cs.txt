using System;
public static class Probe
{
    static string log = "";
    static int Note(string s) { log += s; return 0; }
    static void Set(ref int slot, int value) { slot = value; }
    public static string Run()
    {
        var arr = new int[2];
        try { Set(ref arr[5], Note("second")); }
        catch (IndexOutOfRangeException) { log += "|oob"; }
        object[] objs = new string[1];
        try { Set2(ref objs[0], Note("third")); }
        catch (ArrayTypeMismatchException) { log += "|covariance"; }
        return log;
    }
    static void Set2(ref object slot, int value) { slot = value; }
}
