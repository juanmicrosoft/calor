using System.Diagnostics.CodeAnalysis;
public static class Probe
{
    static bool TryName(int id, [NotNullWhen(true)] out string? name) { name = id == 1 ? "one" : null; return name != null; }
    public static string Run()
    {
        if (TryName(1, out var n)) { string s = n; return s.Length.ToString(); }
        return "none";
    }
}
