public sealed class Box { public int X = 1; }
public static class Probe
{
    static readonly Box shared = new Box();
    static int calls;
    static Box Get() { calls++; return shared; }
    static void Bump(ref int x) { x = x + 1; }
    public static string Run()
    {
        Bump(ref Get().X);
        return shared.X + "|" + calls;
    }
}
