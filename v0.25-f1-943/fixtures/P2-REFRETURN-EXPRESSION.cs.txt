// 0.25 F1 PR 2 (#943) fixture: an expression-bodied ref return (W1 finding 2) is preserved as C#
// like the block-bodied one (D1 RO-UNS-1); before, its `ref` was stripped and the round trip failed.
public static class Probe
{
    static int slot = 3;

    static ref int First(int[] a) => ref a[0];

    static ref int Last(int[] a)
    {
        return ref a[a.Length - 1];
    }

    static ref int Slot => ref slot;

    public static string Run()
    {
        var a = new[] { 1, 2, 3 };
        First(a) = 7;
        Last(a) = 9;
        Slot = 4;
        return a[0] + "|" + a[2] + "|" + slot;
    }
}
