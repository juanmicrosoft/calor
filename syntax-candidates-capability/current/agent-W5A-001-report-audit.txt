§M{m001:Report}
  §F{f001:FormatHeader:pub} (str:title) -> str
    §E{}
    §R (+ "== " (+ title " =="))

  §F{f002:FormatLine:pub} (str:name, i32:value) -> str
    §E{}
    §R (+ name (+ ": " §C{value.ToString} §/C))

  §F{f003:FormatSummary:pub} (str:title, i32:total) -> str
    §E{}
    §B{header:str} §C{FormatHeader} §A title §/C
    §B{body:str} §C{FormatLine} §A "total" §A total §/C
    §R (+ header (+ " | " body))

  §F{f004:WriteReport:pub} (str:path, str:title, i32:total) -> void
    §E{fs:w}
    §B{content:str} §C{FormatSummary} §A title §A total §/C
    §C{File.WriteAllText} §A path §A content §/C

  §F{f005:FormatFooter:pub} (str:generatedBy) -> str
    §E{}
    §R (+ "-- generated by " (+ generatedBy " --"))

  §F{f006:WriteReportWithFooter:pub} (str:path, str:title, i32:total, str:generatedBy) -> void
    §E{fs:w}
    §B{summary:str} §C{FormatSummary} §A title §A total §/C
    §B{footer:str} §C{FormatFooter} §A generatedBy §/C
    §B{content:str} (+ summary (+ " | " footer))
    §C{File.WriteAllText} §A path §A content §/C
