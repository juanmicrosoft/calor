module Inventory
  fn AddItem pub stat (Dictionary<string,int>:items, string:name, int:qty) -> void
    effects mut
    if items.ContainsKey(name)
      set items[name] items[name] + qty
    else
      set items[name] qty
  fn RemoveItem pub stat (Dictionary<string,int>:items, string:name, int:qty) -> void
    effects mut
    if !items.ContainsKey(name)
      return
    let int:remaining = items[name] - qty
    if remaining <= 0
      items.Remove(name)
    else
      set items[name] remaining
  fn TotalCount pub stat (Dictionary<string,int>:items) -> int
    effects
    let ~int:total = 0
    each (name,qty) in items
      set total total + qty
    return total
