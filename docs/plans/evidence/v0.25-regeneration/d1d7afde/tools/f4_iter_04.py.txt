#!/usr/bin/env python3
"""0.25 re-proof: F4-ITER-04 measurement on the candidate (scope note: "The three historical
FluentValidation modules (EmptyTester, NotEmptyTester, ITestValidationContinuation) through the
pinned CLI"). Run from the root of a fresh clone at the candidate with the corpus submodules
initialized, after download-z3.sh and a Release build. Reuses the R0 driver (imported unchanged).

For each module: `calor convert` default (the CLI's own defaults, JSON envelope), `calor convert
--no-fallback`, and MCP `calor_convert` default and passthroughOnError. Each CLI output is compiled
with default options (and with --permissive-effects when only Calor04xx fail). The modules depend on
FluentValidation types (project references), so the generated C# is not executed; the outcome is the
conversion outcome: preserved = iterator accessor kept as §CSHARP member interop, no lossy
substitution, no drop, and no native §YIELD in an accessor (Calor0209)."""
import importlib.util
import json
import os
import re
import sys

ROOT = os.getcwd()
spec = importlib.util.spec_from_file_location("r0", os.path.join(ROOT, "docs/plans/evidence/v0.25-r0-1426/reproduce/reproduce.py"))
r0 = importlib.util.module_from_spec(spec)
spec.loader.exec_module(r0)

MODULES = {
    "EmptyTester": "bench/corpus/FluentValidation/src/FluentValidation.Tests/EmptyTester.cs",
    "NotEmptyTester": "bench/corpus/FluentValidation/src/FluentValidation.Tests/NotEmptyTester.cs",
    "ITestValidationContinuation": "bench/corpus/FluentValidation/src/FluentValidation/TestHelper/ITestValidationContinuation.cs",
}


def outcome(calr, losses, lossy, drops):
    if calr is None:
        return "rejected"
    if lossy or drops:
        return "lossy"
    interop = "§CSHARP" in calr or "§CS{" in calr
    return "preserved" if interop else "native"


def main():
    out_dir = sys.argv[1]
    os.makedirs(out_dir, exist_ok=True)
    result = {"schema": "v0.25-reproof/f4-iter-04/1", "modules": {}}
    sources = {}
    for name, path in MODULES.items():
        if not os.path.exists(path):
            print(f"missing {path}", file=sys.stderr)
            return 2
        sources[name] = open(path, encoding="utf-8").read()
        rec = {"path": path, "sha256": r0.sha(path), "surfaces": {}}
        for mode, extra in (("cli-default", []), ("cli-no-fallback", ["--no-fallback"])):
            calr_path = os.path.join(out_dir, f"{name}.{mode}.calr")
            if os.path.exists(calr_path):
                os.remove(calr_path)
            code, so, se = r0.run(["convert", path, "-o", calr_path, "--format", "json"] + extra)
            env = r0.summarize_envelope(so)
            calr = open(calr_path, encoding="utf-8").read() if code == 0 and os.path.exists(calr_path) else None
            s = {"convertExit": code, **env,
                 "csharpInteropBlocks": len(re.findall(r"§CSHARP", calr)) if calr else None,
                 "yieldInGetterTokens": len(re.findall(r"§YIELD", calr)) if calr else None}
            if calr is not None:
                c = r0.compile_with_fallback(calr_path, os.path.join(out_dir, f"{name}.{mode}.g.cs"))
                s["compile"] = c["compile"]
                if "compilePermissiveEffects" in c:
                    s["compilePermissiveEffects"] = c["compilePermissiveEffects"]
            s["outcome"] = outcome(calr, env.get("lossFeatures"), env.get("lossySubstitutionCount"), env.get("dropCount"))
            rec["surfaces"][mode] = s
        result["modules"][name] = rec
    mcp = r0.mcp_convert(list(sources.items()))
    for (name, mode), (is_error, payload) in mcp.items():
        if mode == "passthroughOnError-moduleName":
            continue
        calr = payload.get("calorSource") or payload.get("calor") or payload.get("output") if isinstance(payload, dict) else None
        stats = (payload.get("stats") or {}) if isinstance(payload, dict) else {}
        s = {"isError": is_error, "success": payload.get("success") if isinstance(payload, dict) else None,
             "interopBlocks": len(re.findall(r"§CSHARP", calr)) if isinstance(calr, str) else None,
             "issueFeatures": sorted({i.get("feature") for i in (payload.get("issues") or []) if isinstance(i, dict) and i.get("feature")}) if isinstance(payload, dict) else [],
             "stats": stats,
             "conversionPaths": payload.get("conversionPaths") if isinstance(payload, dict) else None}
        s["outcome"] = "rejected" if is_error or not isinstance(calr, str) else outcome(calr, None, stats.get("lossySubstitutionCount"), stats.get("dropCount"))
        result["modules"][name]["surfaces"]["mcp-" + mode] = s
    result["allPreserved"] = all(s["outcome"] in ("preserved",) for m in result["modules"].values()
                                 for k, s in m["surfaces"].items() if k != "cli-no-fallback")
    json.dump(result, open(os.path.join(out_dir, "f4-iter-04.json"), "w"), indent=1, ensure_ascii=False)
    print(json.dumps({k: {s: v["outcome"] for s, v in m["surfaces"].items()} for k, m in result["modules"].items()}, indent=1))
    print("allPreserved", result["allPreserved"])
    return 0


if __name__ == "__main__":
    sys.exit(main())
