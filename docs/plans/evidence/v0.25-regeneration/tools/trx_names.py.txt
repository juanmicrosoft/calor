import sys, xml.etree.ElementTree as ET, collections
ns = {"t": "http://microsoft.com/schemas/VisualStudio/TeamTest/2010"}
root = ET.parse(sys.argv[1]).getroot()
res = {r.attrib["testName"]: r.attrib["outcome"] for r in root.findall("t:Results/t:UnitTestResult", ns)}
names = collections.Counter()
for n, o in sorted(res.items()):
    base = n.split("(")[0]
    names[(base, o)] += 1
for (b, o), c in sorted(names.items()):
    print(f"{c:4} {o:8} {b}")
