#!/bin/bash
# A1 (#1408) R2 release-gate dry run against the DRAFT terminal record, in a scratch clone only.
# Usage: dryrun.sh <branch commit holding the record> <out dir>
# refs/remotes/origin/main is moved to the branch commit LOCALLY (simulating the sign-off merge,
# which keeps the same record bytes and so the same digest). Nothing is pushed; nothing is published.
source /private/tmp/claude-501/-Users-juanrivera-repos-juanmicrosoft-calor/98372634-e96e-40c2-8b28-3e5d5ca7761f/scratchpad/a1/env.sh
ADJ="$1"; O="$2"; mkdir -p "$O"
D=$C2/r2/dryrun
A=$C2/r3/out/clone-a
cd $D
git fetch -q origin
git fetch -q /Users/juanrivera/repos/juanmicrosoft/calor/.claude/worktrees/agent-a46da240afbe112d5 "$ADJ"
REC=docs/plans/evidence/adjudication-1408/terminal-record.json
if [ -n "$3" ]; then
  # Counterfactual variant: commit it on top of the branch commit in this scratch clone only.
  git checkout -q --detach "$ADJ"
  cp "$3" "$REC"; git add "$REC"
  git -c user.name=a1-dry-run -c user.email=a1-dry-run@invalid -c commit.gpgsign=false commit -q -m "SCRATCH counterfactual record; never pushed"
  ADJ=$(git rev-parse HEAD)
fi
DIG=$(git show "$ADJ:$REC" | shasum -a 256 | cut -d' ' -f1)
ID="calor-adjudication:v1:$ADJ:$DIG"
REAL_MAIN=$(git rev-parse refs/remotes/origin/main)
git update-ref refs/remotes/origin/main "$ADJ"
git checkout -q --detach 04e61f188e85a710a91242e1ddbf35fcd50e329e
{
echo "identity $ID"
echo "real origin/main $REAL_MAIN; simulated main $ADJ; HEAD $(git rev-parse HEAD)"
python3 scripts/verify_release_adjudication.py --identity "$ID" --expect-head --version 0.24.0 \
  --release-notes $A/release-notes.md --nuget-dir $A/build/nupkg --metadata-dir $A/build/release-metadata \
  --website-dir $A/build/website --release-tag v0.24.0 --release-title v0.24.0
echo "SURFACES-EXIT=$?"
} > "$O/gate.txt" 2>&1
# Benchmark surface: write the headline with B2's gate in this checkout, then R2's worktree check.
rm -rf "$O/b1"; mkdir -p "$O/b1"; cp -R $A/b1-regen/. "$O/b1/"
python3 scripts/benchmark_publication_gate.py check --commit "$(git rev-parse HEAD)" --regenerated "$O/b1" --report "$O/b2-report.txt" > "$O/b2.txt" 2>&1
echo "B2-EXIT=$?" >> "$O/gate.txt"
python3 scripts/verify_release_adjudication.py --identity "$ID" --expect-head --version 0.24.0 --benchmark-worktree >> "$O/gate.txt" 2>&1
echo "BENCHMARK-WORKTREE-EXIT=$?" >> "$O/gate.txt"
git checkout -q -- . ; git clean -qfd website/public/data
git update-ref refs/remotes/origin/main "$REAL_MAIN"
rm -rf "$O/b1"
cat "$O/gate.txt"
