public static class Probe
{
    static string log = "";
    static int Next(string tag, int value) { log += tag; return value; }
    static int Div(int a, int b, out int remainder) { remainder = a % b; return a / b; }
    public static string Run()
    {
        int rem;
        int q = Div(remainder: out rem, b: Next("b", 2), a: Next("a", 7));
        return q + "|" + rem + "|" + log;
    }
}
