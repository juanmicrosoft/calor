public struct Point { public int X; }
public static class Probe
{
    static void Bump(ref int x) { x = x + 1; }
    public static string Run()
    {
        var grid = new Point[2, 2];
        Bump(ref grid[1, 1].X);
        var nested = new Point[][] { new Point[] { new Point { X = 5 } } };
        Bump(ref nested[0][0].X);
        return grid[1, 1].X + "|" + nested[0][0].X;
    }
}
