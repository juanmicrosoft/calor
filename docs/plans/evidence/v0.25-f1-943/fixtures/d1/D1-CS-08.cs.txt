public static class Probe
{
    static int BumpAndGet(ref int x) { x = x + 1; return x * 100; }
    public static string Run()
    {
        int v = 1;
        int after = BumpAndGet(ref v) + v;
        int u = 1;
        int before = u + BumpAndGet(ref u);
        return after + "|" + before;
    }
}
