#!/bin/bash
# A1 R2 gate run against the drafted record, in a scratch clone only. Nothing is pushed.
set -uo pipefail
S=/private/tmp/claude-501/-Users-juanrivera-repos-juanmicrosoft-calor/98372634-e96e-40c2-8b28-3e5d5ca7761f/scratchpad
WT=/Users/juanrivera/repos/juanmicrosoft/calor/.claude/worktrees/agent-ab21d2d79a33571b3
D=$S/a1/gate-clone; O=$S/a1/gate-out; N=$S/c26/out/clone-a; A=$N/build
CAND=d1d7afde672691613f7bb34d253e63401ee423ac
URL="https://github.com/juanmicrosoft/calor/pull/1#pullrequestreview-1"
rm -rf "$D" "$O"; mkdir -p "$O"
git clone -q https://github.com/juanmicrosoft/calor.git "$D"
cd "$D"
git fetch -q "$WT" milestone-0.25/a1-1430
git checkout -q --detach FETCH_HEAD
BASE=$(git rev-parse HEAD)
python3 /private/tmp/claude-501/-Users-juanrivera-repos-juanmicrosoft-calor/98372634-e96e-40c2-8b28-3e5d5ca7761f/scratchpad/a1/synth_closures.py . && python3 docs/plans/evidence/adjudication-1430/tools/make_a1_records.py.txt . --approval-url "$URL" > "$O/make.txt"
python3 - <<'EOF'
import json
p = "eng/release-adjudication-map.json"
m = json.load(open(p))
for e in m["releases"]:
    if e["version"] == "0.25.x":
        e["status"] = "adjudicated"
        e["note"] = "Adjudicated by #1430 (terminal record ACCEPTED)."
open(p, "w").write(json.dumps(m, indent=2, ensure_ascii=False) + "\n")
EOF
git add -A docs/plans/evidence eng/release-adjudication-map.json
git -c user.name=a1-scratch -c user.email=a1@invalid -c commit.gpgsign=false commit -q -m "SCRATCH: A1 record with placeholder approval URL; never pushed"
ADJ=$(git rev-parse HEAD)
DIG=$(git show "$ADJ:docs/plans/evidence/adjudication-1430/terminal-record.json" | shasum -a 256 | cut -d' ' -f1)
ID="calor-adjudication:v1:$ADJ:$DIG"
git update-ref refs/remotes/origin/main "$ADJ"
git show "$ADJ:docs/plans/evidence/adjudication-1430/terminal-record.json" > "$O/terminal-record.scratch.json"
git checkout -q --detach "$CAND"
{
echo "base (A1 branch commit) $BASE; scratch adjudication commit $ADJ (local only; origin/main moved locally)"
echo "approval URL placeholder: $URL"
echo "identity $ID"
echo "--- surfaces: notes, packages, metadata, website (clone A of the candidate, re-proof build)"
python3 scripts/verify_release_adjudication.py --identity "$ID" --expect-head --version 0.25.0 \
  --release-notes "$N/release-notes.md" --nuget-dir "$A/nupkg" --metadata-dir "$A/release-metadata" \
  --website-dir "$A/website" --release-tag v0.25.0 --release-title v0.25.0 2>&1
echo "SURFACES-EXIT=$?"
mkdir -p "$O/b1"; cp -R "$N/b1-regen/." "$O/b1/"
python3 scripts/benchmark_publication_gate.py check --commit "$(git rev-parse HEAD)" --regenerated "$O/b1" --report "$O/b2-report.txt" > "$O/b2.txt" 2>&1
echo "B2-EXIT=$?"
git status --porcelain
echo "--- benchmark worktree (record does not approve publication; G015 expected)"
python3 scripts/verify_release_adjudication.py --identity "$ID" --expect-head --version 0.25.0 --benchmark-worktree 2>&1
echo "BENCHMARK-WORKTREE-EXIT=$?"
} > "$O/gate.txt"
rm -rf "$O/b1"
git checkout -q -- . ; git clean -qfd
cat "$O/gate.txt"
