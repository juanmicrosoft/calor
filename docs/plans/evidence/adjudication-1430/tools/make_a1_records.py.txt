usage: make_a1_records.py <repo> --approval-url <URL | PENDING> [--check] [--snapshot-closures]

* claim-registry.json: one claim per substantive statement in the candidate's CHANGELOG.md [0.25.0] section.
  Restatements share an id. Line numbers are 1-based in the rendered notes

  Every file is newly added, so the V025 classifier exempts it (downstream data file, status A).
* With PENDING, it writes only adjudication-1430/draft/terminal-record.pending.json. That is the same record with
  approval-dependent values marked PENDING, for review before approval.
--check regenerates in memory and fails if any committed output differs.

def sha(repo, p):
    return hashlib.sha256((repo / p).read_bytes()).hexdigest()


def lf(repo, p):
    return hashlib.sha256((repo / p).read_bytes().replace(b"\r\n", b"\n")).hexdigest()


def ev(repo, paths):
    out, seen = [], set()
    for p in [NOTES, *paths]:
        if p in seen:
            continue
        seen.add(p)
        if not (repo / p).is_file():
            sys.exit(f"missing evidence file {p}")
        out.append({"path": p, "sha256": sha(repo, p)})
    return out

def check_closures(repo):
    """Final mode needs closures.json: the nine blockers closed (snapshot taken with --snapshot-closures after closing)."""
    if not (repo / CLOSURES).is_file():
        sys.exit(f"{CLOSURES} is missing: close the nine blockers, then run --snapshot-closures")
    snap = json.loads((repo / CLOSURES).read_text())
    got = {i["number"]: i for i in snap.get("issues", [])}
    bad = [n for n in BLOCKERS if got.get(n, {}).get("state") != "CLOSED" or not got.get(n, {}).get("closedAt")]
    if sorted(got) != sorted(BLOCKERS) or bad:
        sys.exit(f"{CLOSURES} must list exactly the nine blockers, each CLOSED with closedAt; not closed: {bad}")


def snapshot_closures(repo):
    import subprocess
    issues = []
    for n in BLOCKERS:
        out = subprocess.run(["gh", "issue", "view", str(n), "-R", "juanmicrosoft/calor", "--json", "number,state,closedAt,title"],
                             capture_output=True, text=True, check=True).stdout
        issues.append(json.loads(out))
    snap = {"schema": "adjudication-1430/closures/1", "issue": 1430, "decision": "D-A1-OUTCOME-ACCEPTED",
            "note": "Snapshot of the nine #1430 closure blockers, taken with gh after the maintainer closed them.", "issues": issues}
    (repo / CLOSURES).write_text(json.dumps(snap, indent=1, ensure_ascii=False) + "\n")
    print(f"wrote {CLOSURES}")


def outputs(repo, url):
    reg = json.dumps(registry(), indent=1, ensure_ascii=False) + "\n"
    out = {f"{A1}/claim-registry.json": reg}
    if url == "PENDING":
        out[f"{A1}/draft/terminal-record.pending.json"] = json.dumps(record(repo, url, {}, reg), indent=1, ensure_ascii=False) + "\n"
    else:
        if not URL_RE.match(url):
            sys.exit(f"approval URL {url!r} is not a pull-request review URL")
        check_closures(repo)
        appr = {f: approved_ready(repo, f, url) for f in FAM}
        for f in FAM:
            out[APPROVED[f]] = appr[f]
        out[f"{A1}/terminal-record.json"] = json.dumps(record(repo, url, appr, reg), indent=1, ensure_ascii=False) + "\n"
    return out


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("repo")
    ap.add_argument("--approval-url", required=True)
    ap.add_argument("--check", action="store_true")
    ap.add_argument("--snapshot-closures", action="store_true", help="write closures.json from gh (once, after closing the nine issues)")
    a = ap.parse_args()
    repo = Path(a.repo)
    if a.snapshot_closures:
        snapshot_closures(repo)
        return 0
    bad = 0
    for path, text in outputs(repo, a.approval_url).items():
        p = repo / path
        if a.check:
            if not p.is_file() or p.read_text() != text:
                print(f"DIFFERS {path}")
                bad += 1
        else:
            p.parent.mkdir(parents=True, exist_ok=True)
            p.write_text(text)
            print(f"wrote {path} {hashlib.sha256(text.encode()).hexdigest()}")
    if a.check:
        print("CHECK", "FAIL" if bad else "OK")
    return 1 if bad else 0


if __name__ == "__main__":
    sys.exit(main())
