// 0.25 R0 (#1426) baseline fixture F3-LOCAL-04 (#847): shapes outside the frozen native slice
// (capturing, generic, static-with-capture-free iterator). They must stay preserved, never
// counted as native support.
using System;
using System.Collections.Generic;

public static class Probe
{
    public static int Capturing(int k)
    {
        int AddK(int x) => x + k;
        return AddK(1);
    }

    public static string Generic()
    {
        T Echo<T>(T value) => value;
        return Echo("g") + Echo(3);
    }

    public static int IteratorLocal()
    {
        IEnumerable<int> Twice()
        {
            yield return 1;
            yield return 1;
        }
        int n = 0;
        foreach (var x in Twice())
        {
            n = n + x;
        }
        return n;
    }

    public static string Run()
    {
        return Capturing(4) + "|" + Generic() + "|" + IteratorLocal();
    }
}
