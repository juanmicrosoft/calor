    §CSHARP{public class Grid
{
    public Cell[,] Build()

            { new Cell { V = 1 }, new Cell { V = 2 } },
            { new Cell { V = 3 }, new Cell { V = 4 } }
        };
    }
}}§/CSHARP

    §CSHARP{public static class Probe
{

    {
        var cells = new Grid().Build();

    }
}}§/CSHARP
