    private const string Schema = "calor.issue1400.nominal-policy.migration-feasibility.source.v1";
    private const string FixtureNamespace = "MigrationFeasibility.NominalPolicy1400";
    private const string ItemMetadataName = FixtureNamespace + ".Item";
    private const string NativeSurfaceMetadataName = FixtureNamespace + ".NativeSurface";
    private const string DownstreamMetadataName = FixtureNamespace + ".Downstream";
    private const string RuntimeProbeMethodName = "RunAll";

    private static readonly JsonSerializerOptions JsonOptions = new()
    {
        WriteIndented = true,
        DefaultIgnoreCondition = JsonIgnoreCondition.WhenWritingNull,
    };

    public static int Main(string[] args)
    {
        EvidenceReport? report = null;
        try
        {
            var options = DriverOptions.Parse(args);
            report = Execute(options);
            WriteReport(report, options.OutputPath);
            return report.HardFailures.Count == 0 ? 0 : 2;
        }
        catch (Exception ex) when (ex is not OutOfMemoryException)
        {
            report ??= new EvidenceReport { Schema = Schema, GeneratedAtUtc = DateTimeOffset.UtcNow };
            report.HardFailures.Add(new FailureRecord("driver-unhandled-exception", ex.GetType().FullName ?? ex.GetType().Name, ex.ToString()));
            WriteReport(report, DriverOptions.TryGetOutputPath(args));
            return 1;
        }
    }

    private static EvidenceReport Execute(DriverOptions options)
    {
        var report = new EvidenceReport
        {
            Schema = Schema,
            GeneratedAtUtc = DateTimeOffset.UtcNow,
            ConfigurationLabel = options.ConfigurationLabel,
            Inputs = new InputRecord(
                CalorRoot: Path.GetFullPath(options.CalorRoot),
                OriginalCSharpSha256: Sha256Text(FixtureSources.NativeReferencePassthrough),
                OriginalCSharp: FixtureSources.NativeReferencePassthrough,
                ExpectedFileHashes: options.ExpectedFileHashes),
            Driver = DriverRecord.Capture(),
            SourceContext = CaptureSourceContext(options),
            ReferenceProfile = CaptureReferenceProfile(),
        };

        ValidateExpectedSourceHashes(report, options);

        var originalRoslyn = AnalyzeOriginalRoslyn(FixtureSources.NativeReferencePassthrough);
        report.OriginalRoslyn = originalRoslyn;
        var originalRuntime = CompileAndRunCSharp("original-csharp", FixtureSources.NativeReferencePassthrough, expectRuntimeShape: true);
        report.OriginalRuntime = originalRuntime;
        if (!originalRuntime.RoslynEmitSuccess || !originalRuntime.RuntimeExpectationSatisfied)
        {
            report.HardFailures.Add(new FailureRecord(
                "original-runtime",
                "Original C# fixture did not compile/run as expected.",
                JsonSerializer.Serialize(originalRuntime, JsonOptions)));
        }

        var conversion = ConvertOriginal();
        report.Conversion = conversion;

        CalorVariantRecord? currentVariant = null;
        if (conversion.CalorSource is not null)
        {
            currentVariant = AnalyzeCalorVariant("current-converter", conversion.CalorSource, expectRuntimeShape: false);
            report.CurrentVariant = currentVariant;
            report.CurrentConverterClassification = ClassifyCurrentConverter(originalRoslyn, currentVariant);
        }
        else
        {
            report.CurrentConverterClassification = new ConverterClassification(
                HasConvertedCalor: false,
                NativeDeclarationCount: 0,
                InteropDeclarationCount: 0,
                EchoParameterDeclaredNullable: null,
                EchoReturnDeclaredNullable: null,
                FaithfulCandidateMayBeAttempted: false,
                Reason: "Current converter returned no Calor source; candidate AST representation is blocked.");
        }

        var identityGate = BuildIdentityGate(originalRoslyn, conversion, currentVariant);
        report.IdentityGate = identityGate;

        if (identityGate.CanAttemptFaithfulCandidate && conversion.CalorSource is not null)
        {
            var faithfulCandidate = TryBuildNullableCandidate(conversion.CalorSource, identityGate);
            report.FaithfulCandidate = faithfulCandidate;
            if (faithfulCandidate.CandidateCalorSource is not null)
            {
                var faithfulVariant = AnalyzeCalorVariant(
                    "fixture-specific-nullable-reference-candidate",
                    faithfulCandidate.CandidateCalorSource,
                    expectRuntimeShape: true);
                report.FaithfulVariant = faithfulVariant;
                if (!faithfulVariant.DefaultCompile.Accepted || faithfulVariant.Runtime?.RuntimeExpectationSatisfied != true)
                {
                    var reason = "Candidate did not compile and run with expected null/present identity behavior under default Program.Compile.";
                    faithfulCandidate = faithfulCandidate with
                    {
                        Status = "failed",
                        FailureReason = reason,
                    };
                    report.FaithfulCandidate = faithfulCandidate;
                    report.HardFailures.Add(new FailureRecord(
                        "faithful-candidate-runtime",
                        reason,
                        JsonSerializer.Serialize(faithfulVariant, JsonOptions)));
                }
                else
                {
                    faithfulCandidate = faithfulCandidate with { Status = "passed-fixture-only" };
                    report.FaithfulCandidate = faithfulCandidate;
                }
            }
        }
        else
        {
            report.FaithfulCandidate = FaithfulCandidateRecord.Skipped(identityGate.BlockingReason ?? "Identity gate failed.");
        }

        if (report.FaithfulCandidate?.Status != "passed-fixture-only")
        {
            var rawInteropFallback = BuildRawInteropFallback();
            report.RawInteropFallback = rawInteropFallback;
            var rawInteropVariant = AnalyzeCalorVariant(
                "explicit-raw-csharp-interop-fallback",
                rawInteropFallback.CalorSource,
                expectRuntimeShape: true);
            report.RawInteropVariant = rawInteropVariant;
            if (!rawInteropVariant.DefaultCompile.Accepted || rawInteropVariant.Runtime?.RuntimeExpectationSatisfied != true)
            {
                report.HardFailures.Add(new FailureRecord(
                    "raw-interop-fallback-runtime",
                    "Explicit §CSHARP fallback did not preserve expected fixture behavior.",
                    JsonSerializer.Serialize(rawInteropVariant, JsonOptions)));
            }
        }

        report.Overall = Summarize(report);
        return report;
    }

    private static ConversionRecord ConvertOriginal()
    {
        var options = new ConversionOptions
        {
            ModuleName = "NominalReferenceMigrationFixture",
            Fidelity = ConversionFidelity.Lossless,
            AutoGenerateIds = true,
            GracefulFallback = true,
            ValidateRoundTripCSharp = true,
            OutputKind = OutputKind.DynamicallyLinkedLibrary,
            ParseOptions = CSharpParseOptions.Default
                .WithLanguageVersion(LanguageVersion.Preview)
                .WithDocumentationMode(DocumentationMode.Parse),
        };
        var converter = new CSharpToCalorConverter(options);
        var result = converter.Convert(FixtureSources.NativeReferencePassthrough, "NativeReferenceMigrationFixture.cs");
        return new ConversionRecord(
            Success: result.Success,
            CalorSource: result.CalorSource,
            CalorSha256: result.CalorSource is null ? null : Sha256Text(result.CalorSource),
            Issues: result.Issues.Select(i => new ConversionIssueRecord(i.Severity.ToString(), i.Message, i.Feature, i.Line, i.Column)).ToList(),
            Losses: result.Losses.Select(l => l.ToString()).ToList(),
            NativeConversionCount: result.NativeConversionCount,
            InteropPreservationCount: result.InteropPreservationCount,
            LossySubstitutionCount: result.LossySubstitutionCount,
            DropCount: result.DropCount,
            DurationMilliseconds: result.Duration.TotalMilliseconds);
    }

    private static CalorVariantRecord AnalyzeCalorVariant(string name, string calorSource, bool expectRuntimeShape)
    {
        var parse = ParseCalor(calorSource);
        BindingRecord? binding = null;
        if (parse.Module is not null)
            binding = BindCalor(parse.Module, $"{name}.calr");

        var defaultCompile = CompileCalor(name, calorSource, enforceEffects: true);
        CompileRecord? effectsDisabled = null;
        if (!defaultCompile.Accepted && defaultCompile.Diagnostics.Any(d => d.Code.StartsWith("Calor04", StringComparison.Ordinal)))
            effectsDisabled = CompileCalor(name + "-effects-disabled", calorSource, enforceEffects: false);

        RuntimeRecord? runtime = null;
        var runtimeSource = defaultCompile.Accepted
            ? defaultCompile.GeneratedCSharp
            : effectsDisabled?.Accepted == true ? effectsDisabled.GeneratedCSharp : null;
        if (runtimeSource is not null)
        {
            runtime = CompileAndRunCSharp(
                name,
                runtimeSource,
                expectRuntimeShape,
                runtimeSourceKind: defaultCompile.Accepted ? "default-Program.Compile" : "effects-disabled-Program.Compile");
        }

        return new CalorVariantRecord(
            Name: name,
            CalorSource: calorSource,
            CalorSha256: Sha256Text(calorSource),
            NativeCoverage: parse.Module is null ? null : CaptureNativeCoverage(parse.Module),
            Parse: parse.ToRecord(),
            Binding: binding,
            DefaultCompile: defaultCompile,
            EffectsDisabledCompile: effectsDisabled,
            Runtime: runtime);
    }

    private static ParseAttempt ParseCalor(string calorSource)
    {
        var diagnostics = new DiagnosticBag();
        diagnostics.SetFilePath("candidate.calr");
        try
        {
            var lexer = new Lexer(calorSource, diagnostics);
            var tokens = lexer.TokenizeAllForParser();
            if (diagnostics.HasErrors)
                return new ParseAttempt(null, Diagnostics(diagnostics), null);
            var parser = new Parser(tokens, diagnostics);
            var module = parser.Parse();
            return new ParseAttempt(module, Diagnostics(diagnostics), null);
        }
        catch (Exception ex) when (ex is not OutOfMemoryException)
        {
            return new ParseAttempt(null, Diagnostics(diagnostics), ExceptionRecord.From(ex));
        }
    }

    private static BindingRecord BindCalor(ModuleNode module, string sourceIdentity)
    {
        var diagnostics = new DiagnosticBag();
        diagnostics.SetFilePath(sourceIdentity);
        try
        {
            var binder = new Binder(diagnostics, sourceIdentity);
            var bound = binder.Bind(module);
            return new BindingRecord(
                Success: !diagnostics.HasErrors,
                Diagnostics: Diagnostics(diagnostics),
                BinderMetadataProfile: CaptureBinderMetadataProfile(binder),
                BoundIdentity: CaptureBoundIdentity(bound),
                Exception: null);
        }
        catch (Exception ex) when (ex is not OutOfMemoryException)
        {
            return new BindingRecord(
                Success: false,
                Diagnostics: Diagnostics(diagnostics),
                BinderMetadataProfile: null,
                BoundIdentity: null,
                Exception: ExceptionRecord.From(ex));
        }
    }

    private static CompileRecord CompileCalor(string name, string calorSource, bool enforceEffects)
    {
        try
        {
            var result = CalorProgram.Compile(
                calorSource,
                name + ".calr",
                new CompilationOptions
                {
                    EnforceEffects = enforceEffects,
                    StatusWriter = TextWriter.Null,
                });
            return new CompileRecord(
                Label: enforceEffects ? "default" : "effects-disabled",
                EnforceEffects: enforceEffects,
                Accepted: !result.HasErrors && !string.IsNullOrWhiteSpace(result.GeneratedCode),
                Diagnostics: Diagnostics(result.Diagnostics),
                GeneratedCSharp: string.IsNullOrWhiteSpace(result.GeneratedCode) ? null : result.GeneratedCode,
                GeneratedCSharpSha256: string.IsNullOrWhiteSpace(result.GeneratedCode) ? null : Sha256Text(result.GeneratedCode),
                Exception: null);
        }
        catch (Exception ex) when (ex is not OutOfMemoryException)
        {
            return new CompileRecord(
                Label: enforceEffects ? "default" : "effects-disabled",
                EnforceEffects: enforceEffects,
                Accepted: false,
                Diagnostics: [],
                GeneratedCSharp: null,
                GeneratedCSharpSha256: null,
                Exception: ExceptionRecord.From(ex));
        }
    }

    private static RuntimeRecord CompileAndRunCSharp(
        string name,
        string csharpSource,
        bool expectRuntimeShape,
        string runtimeSourceKind = "original-csharp")
    {
        var parseOptions = CSharpParseOptions.Default.WithLanguageVersion(LanguageVersion.Preview);
        var tree = CSharpSyntaxTree.ParseText(csharpSource, parseOptions, $"{name}.cs", Encoding.UTF8);
        var syntaxDiagnostics = tree.GetDiagnostics().Where(IsRoslynError).Select(RoslynDiagnosticRecord.From).ToList();
        var compilation = CSharpCompilation.Create(
            "NominalMigrationFixture_" + SanitizeAssemblyName(name) + "_" + Guid.NewGuid().ToString("N"),
            [tree],
            GeneratedCSharpCompiler.References,
            new CSharpCompilationOptions(
                OutputKind.DynamicallyLinkedLibrary,
                allowUnsafe: true,
                nullableContextOptions: NullableContextOptions.Enable));

        using var image = new MemoryStream();
        var emit = compilation.Emit(image);
        var emitDiagnostics = emit.Diagnostics.Where(IsRoslynError).Select(RoslynDiagnosticRecord.From).ToList();
        if (syntaxDiagnostics.Count > 0 || !emit.Success)
        {
            return new RuntimeRecord(
                SourceKind: runtimeSourceKind,
                InputCSharpSha256: Sha256Text(csharpSource),
                RoslynEmitSuccess: false,
                SyntaxDiagnostics: syntaxDiagnostics,
                EmitDiagnostics: emitDiagnostics,
                EntryPointFound: false,
                Runtime: null,
                RuntimeExpectationSatisfied: !expectRuntimeShape,
                Exception: null);
        }

        try
        {
            image.Position = 0;
            var alc = new AssemblyLoadContext("NominalMigrationFixture_" + Guid.NewGuid().ToString("N"), isCollectible: true);
            try
            {
                var assembly = alc.LoadFromStream(image);
                var method = assembly.GetTypes()
                    .Select(type => type.GetMethod(RuntimeProbeMethodName, BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Static))
                    .SingleOrDefault(method => method is not null && method.GetParameters().Length == 0);
                if (method is null)
                {
                    return new RuntimeRecord(runtimeSourceKind, Sha256Text(csharpSource), true, syntaxDiagnostics, emitDiagnostics, false, null, !expectRuntimeShape, null);
                }

                object? returned;
                try
                {
                    returned = method.Invoke(null, null);
                }
                catch (TargetInvocationException tie) when (tie.InnerException is not null)
                {
                    return new RuntimeRecord(
                        runtimeSourceKind,
                        Sha256Text(csharpSource),
                        true,
                        syntaxDiagnostics,
                        emitDiagnostics,
                        true,
                        null,
                        false,
                        ExceptionRecord.From(tie.InnerException));
                }

                var values = CapturePublicValues(returned);
                var expectation = !expectRuntimeShape || RuntimeShapeSatisfied(values);
                return new RuntimeRecord(runtimeSourceKind, Sha256Text(csharpSource), true, syntaxDiagnostics, emitDiagnostics, true, values, expectation, null);
            }
            finally
            {
                alc.Unload();
            }
        }
        catch (Exception ex) when (ex is not OutOfMemoryException)
        {
            return new RuntimeRecord(runtimeSourceKind, Sha256Text(csharpSource), true, syntaxDiagnostics, emitDiagnostics, true, null, false, ExceptionRecord.From(ex));
        }
    }

    private static OriginalRoslynRecord AnalyzeOriginalRoslyn(string source)
    {
        var parseOptions = CSharpParseOptions.Default.WithLanguageVersion(LanguageVersion.Preview);
        var tree = CSharpSyntaxTree.ParseText(source, parseOptions, "NativeReferenceMigrationFixture.cs", Encoding.UTF8);
        var compilation = CSharpCompilation.Create(
            "OriginalDeclarationIdentity",
            [tree],
            GeneratedCSharpCompiler.References,
            new CSharpCompilationOptions(OutputKind.DynamicallyLinkedLibrary, nullableContextOptions: NullableContextOptions.Disable));
        var model = compilation.GetSemanticModel(tree, ignoreAccessibility: true);
        var root = tree.GetCompilationUnitRoot();
        var item = compilation.GetTypeByMetadataName(ItemMetadataName);
        var native = compilation.GetTypeByMetadataName(NativeSurfaceMetadataName);
        var downstream = compilation.GetTypeByMetadataName(DownstreamMetadataName);

        var declarations = new List<RoslynDeclarationRecord>();
        if (item is not null)
            declarations.Add(RoslynDeclarationRecord.FromType(item, model, root));
        foreach (var methodName in new[] { "Echo", "MakeFresh" })
            declarations.Add(CaptureMethod(native, methodName, item, model, root));
        foreach (var methodName in new[] { "Receive", "ReadLabelLength" })
            declarations.Add(CaptureMethod(downstream, methodName, item, model, root));

        var diagnostics = compilation.GetDiagnostics().Where(IsRoslynError).Select(RoslynDiagnosticRecord.From).ToList();
        return new OriginalRoslynRecord(
            CompilationSuccess: diagnostics.Count == 0,
            Diagnostics: diagnostics,
            ItemSymbol: item is null ? null : RoslynSymbolRecord.From(item),
            Declarations: declarations);
    }

    private static RoslynDeclarationRecord CaptureMethod(INamedTypeSymbol? container, string methodName, INamedTypeSymbol? item, SemanticModel model, CompilationUnitSyntax root)
    {
        if (container is null)
            return RoslynDeclarationRecord.Missing(methodName, "container type missing");
        var method = container.GetMembers(methodName).OfType<IMethodSymbol>().SingleOrDefault();
        if (method is null)
            return RoslynDeclarationRecord.Missing(methodName, "method missing");
        var syntax = method.DeclaringSyntaxReferences.Select(r => r.GetSyntax()).OfType<MethodDeclarationSyntax>().SingleOrDefault();
        var returnMatchesItem = item is not null && SymbolEqualityComparer.Default.Equals(method.ReturnType, item);
        var parameters = method.Parameters.Select(parameter => new RoslynParameterRecord(
            parameter.Name,
            parameter.Type.ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat),
            parameter.Type.NullableAnnotation.ToString(),
            item is not null && SymbolEqualityComparer.Default.Equals(parameter.Type, item),
            parameter.DeclaringSyntaxReferences.Select(r => r.GetSyntax()).OfType<ParameterSyntax>().SingleOrDefault() is { } ps
                ? model.GetNullableContext(ps.Type?.SpanStart ?? ps.SpanStart).ToString()
                : "unavailable",
            parameter.DeclaringSyntaxReferences.Select(r => r.GetSyntax()).OfType<ParameterSyntax>().SingleOrDefault()?.Type is NullableTypeSyntax)).ToList();
        return new RoslynDeclarationRecord(
            Kind: "method",
            Name: methodName,
            MetadataName: method.ContainingType.ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat) + "." + method.MetadataName,
            NullableContextAtDeclaration: syntax is null ? "unavailable" : model.GetNullableContext(syntax.SpanStart).ToString(),
            SyntaxTypeText: syntax?.ReturnType.ToString(),
            SyntaxTypeIsNullable: syntax?.ReturnType is NullableTypeSyntax,
            Symbol: RoslynSymbolRecord.From(method),
            ReturnType: new RoslynTypeRecord(
                method.ReturnType.ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat),
                method.ReturnType.NullableAnnotation.ToString(),
                returnMatchesItem,
                syntax is null ? "unavailable" : model.GetNullableContext(syntax.ReturnType.SpanStart).ToString(),
                syntax?.ReturnType is NullableTypeSyntax),
            Parameters: parameters,
            MissingReason: null);
    }

    private static IdentityGateRecord BuildIdentityGate(OriginalRoslynRecord original, ConversionRecord conversion, CalorVariantRecord? current)
    {
        if (!original.CompilationSuccess)
            return IdentityGateRecord.Blocked("Original Roslyn fixture did not compile.");
        if (original.ItemSymbol is null)
            return IdentityGateRecord.Blocked("Original Item symbol is unavailable.");
        if (conversion.CalorSource is null)
            return IdentityGateRecord.Blocked("No converted Calor source is available.");
        if (current is null)
            return IdentityGateRecord.Blocked("Current converter variant is unavailable.");
        var currentBinding = current.Binding;
        if (current.Parse.Success != true || currentBinding?.Success != true)
            return IdentityGateRecord.Blocked("Converted Calor did not parse and bind cleanly.");

        var echo = original.Declarations.SingleOrDefault(d => d.Name == "Echo");
        var receive = original.Declarations.SingleOrDefault(d => d.Name == "Receive");
        if (!IsGenuinelyUnannotatedItemPassthrough(echo, requireParameter: true))
            return IdentityGateRecord.Blocked("Echo is not proven as a nullable-disabled unannotated Item parameter/return passthrough declaration.");
        if (!IsGenuinelyUnannotatedItemPassthrough(receive, requireParameter: true))
            return IdentityGateRecord.Blocked("Downstream.Receive is not proven as a nullable-disabled unannotated Item parameter/return declaration.");

        var calorEcho = FindMethodShape(current, "Echo");
        var calorReceive = FindMethodShape(current, "Receive");
        if (calorEcho is null || calorReceive is null)
            return IdentityGateRecord.Blocked("Converted Calor AST/bound identity did not expose Echo and Receive methods.");
        if (!calorEcho.HasItemParameterAndReturn || !calorReceive.HasItemParameterAndReturn)
            return IdentityGateRecord.Blocked("Converted Calor method declarations do not retain Item parameter/return shapes needed for a fixture-specific transform.");
        var boundItems = currentBinding!.BoundIdentity?.Types
            .Where(t => t.Name == "Item" || t.QualifiedName.EndsWith(".Item", StringComparison.Ordinal))
            .ToList() ?? [];
        if (boundItems.Count != 1 || !boundItems[0].IsReferenceType)
            return IdentityGateRecord.Blocked($"Binder did not expose exactly one reference TypeSymbol for Item in the converted Calor program; observed {boundItems.Count}.");
        var boundItem = boundItems[0];

        return new IdentityGateRecord(
            CanAttemptFaithfulCandidate: true,
            BlockingReason: null,
            ProvenDeclarations: ["NativeSurface.Echo", "Downstream.Receive"],
            Evidence: new Dictionary<string, object?>
            {
                ["originalItemSymbol"] = original.ItemSymbol,
                ["echoRoslyn"] = echo,
                ["receiveRoslyn"] = receive,
                ["echoCalor"] = calorEcho,
                ["receiveCalor"] = calorReceive,
                ["boundItemTypeSymbol"] = boundItem,
            });
    }

    private static bool IsGenuinelyUnannotatedItemPassthrough(RoslynDeclarationRecord? method, bool requireParameter)
    {
        if (method?.ReturnType is null) return false;
        if (!method.ReturnType.MatchesItemSymbol) return false;
        if (method.ReturnType.NullableAnnotation != nameof(Microsoft.CodeAnalysis.NullableAnnotation.None)) return false;
        if (method.ReturnType.SyntaxTypeIsNullable == true) return false;
        if (!method.ReturnType.NullableContext.Contains(nameof(NullableContext.Disabled), StringComparison.Ordinal)) return false;
        if (!requireParameter) return true;
        var parameter = method.Parameters.SingleOrDefault();
        return parameter is not null
            && parameter.MatchesItemSymbol
            && parameter.NullableAnnotation == nameof(Microsoft.CodeAnalysis.NullableAnnotation.None)
            && parameter.SyntaxTypeIsNullable != true
            && parameter.NullableContext.Contains(nameof(NullableContext.Disabled), StringComparison.Ordinal);
    }

    private static MethodShapeRecord? FindMethodShape(CalorVariantRecord variant, string methodName)
    {
        var declared = variant.NativeCoverage?.MethodDeclarations.SingleOrDefault(m => m.Name == methodName);
        if (declared is null) return null;
        var bound = variant.Binding?.BoundIdentity?.Functions.SingleOrDefault(f => f.Name == methodName);
        var hasItemParameter = declared.Parameters.Count == 1 && StripNullablePrefix(declared.Parameters[0].TypeName) == "Item";
        var hasItemReturn = StripNullablePrefix(declared.OutputTypeName) == "Item";
        return new MethodShapeRecord(
            methodName,
            declared.ContainingType,
            declared.OutputTypeName,
            declared.Parameters,
            hasItemParameter && hasItemReturn,
            bound);
    }

    private static FaithfulCandidateRecord TryBuildNullableCandidate(string currentCalor, IdentityGateRecord gate)
    {
        var parse = ParseCalor(currentCalor);
        if (parse.Module is null)
            return FaithfulCandidateRecord.Failed("Current Calor could not be parsed for AST transformation.");

        try
        {
            var transformer = new FixtureNullableAstTransformer(new HashSet<string>(StringComparer.Ordinal) { "Echo", "Receive" });
            var transformed = transformer.TransformModule(parse.Module);
            if (transformer.ChangedSites.Count != 4)
                return FaithfulCandidateRecord.Failed($"Expected exactly four declaration-site changes (Echo param/return and Receive param/return), observed {transformer.ChangedSites.Count}.");

            var emitted = new CalorEmitter().Emit(transformed);
            return new FaithfulCandidateRecord(
                Status: "attempted",
                FailureReason: null,
                ChangedSites: transformer.ChangedSites,
                CandidateCalorSource: emitted,
                CandidateCalorSha256: Sha256Text(emitted),
                GateEvidence: gate.Evidence);
        }
        catch (Exception ex) when (ex is not OutOfMemoryException)
        {
            return FaithfulCandidateRecord.Failed(ex.ToString());
        }
    }

    private static RawInteropFallbackRecord BuildRawInteropFallback()
    {
        var source = "§M{m_raw:RawInteropFallback}\n"
            + "  §CSHARP{/*__CALOR_COMPILATION_UNIT_PASSTHROUGH__*/\n"
            + FixtureSources.NativeReferencePassthrough.TrimEnd()
            + "\n}§/CSHARP\n";
        var original = CaptureOriginalNativeCoverage(FixtureSources.NativeReferencePassthrough);
        return new RawInteropFallbackRecord(
            CalorSource: source,
            CalorSha256: Sha256Text(source),
            NativeCoverageLoss: new NativeCoverageLossRecord(
                OriginalTypeDeclarations: original.TypeDeclarations,
                OriginalMethodLikeDeclarations: original.MethodLikeDeclarations,
                NativeCalorTypeDeclarations: 0,
                NativeCalorMethodLikeDeclarations: 0,
                LostTypeDeclarations: original.TypeDeclarations,
                LostMethodLikeDeclarations: original.MethodLikeDeclarations,
                Explanation: "Whole-compilation-unit §CSHARP fallback preserves runtime behavior but supplies no native Calor declarations for this fixture."));
    }

    private static ConverterClassification ClassifyCurrentConverter(OriginalRoslynRecord original, CalorVariantRecord? current)
    {
        if (current is null)
            return new ConverterClassification(false, 0, 0, null, null, false, "No current variant.");
        var echo = FindMethodShape(current, "Echo");
        return new ConverterClassification(
            HasConvertedCalor: true,
            NativeDeclarationCount: current.NativeCoverage?.NativeDeclarationCount ?? 0,
            InteropDeclarationCount: current.NativeCoverage?.InteropDeclarationCount ?? 0,
            EchoParameterDeclaredNullable: echo?.Parameters.SingleOrDefault()?.TypeName.StartsWith('?', StringComparison.Ordinal),
            EchoReturnDeclaredNullable: echo?.OutputTypeName?.StartsWith('?', StringComparison.Ordinal),
            FaithfulCandidateMayBeAttempted: echo?.HasItemParameterAndReturn == true,
            Reason: "Current output is classified before any fixture-specific representation. Non-null Item declaration types here are recorded as bare-converter annotation loss when original Roslyn nullable annotation is None under #nullable disable.");
    }

    private static NativeCoverageRecord CaptureNativeCoverage(ModuleNode module)
    {
        var methods = new List<MethodDeclarationRecord>();
        var typeCount = 0;
        var constructorCount = 0;
        void VisitClass(ClassDefinitionNode cls, string? container)
        {
            typeCount++;
            var name = container is null ? cls.Name : container + "." + cls.Name;
            constructorCount += cls.Constructors.Count;

            var methodItems = cls.Items.Count > 0
                ? cls.Items.OfType<MethodNode>()
                : cls.Methods;
            foreach (var method in methodItems)
            {
                methods.Add(new MethodDeclarationRecord(
                    Name: method.Name,
                    ContainingType: name,
                    OutputTypeName: method.Output?.TypeName,
                    Parameters: method.Parameters.Select(ParameterDeclarationRecord.From).ToList()));
            }

            var nestedItems = cls.Items.Count > 0
                ? cls.Items.OfType<ClassDefinitionNode>()
                : cls.NestedClasses;
            foreach (var nested in nestedItems)
                VisitClass(nested, name);
        }
        var moduleClasses = module.Items.Count > 0
            ? module.Items.OfType<ClassDefinitionNode>()
            : module.Classes;
        foreach (var cls in moduleClasses)
            VisitClass(cls, null);
        foreach (var function in module.Functions)
        {
            methods.Add(new MethodDeclarationRecord(function.Name, null, function.Output?.TypeName, function.Parameters.Select(ParameterDeclarationRecord.From).ToList()));
        }
        return new NativeCoverageRecord(
            TypeDeclarations: typeCount,
            MethodDeclarations: methods.DistinctBy(m => (m.ContainingType, m.Name, m.OutputTypeName)).ToList(),
            ConstructorDeclarations: constructorCount,
            InteropBlocks: module.InteropBlocks.Count + module.Classes.Sum(c => c.InteropBlocks.Count),
            NativeDeclarationCount: typeCount + constructorCount + methods.Count,
            InteropDeclarationCount: module.InteropBlocks.Count + module.Classes.Sum(c => c.InteropBlocks.Count));
    }

    private static OriginalCoverageCount CaptureOriginalNativeCoverage(string source)
    {
        var root = CSharpSyntaxTree.ParseText(source, CSharpParseOptions.Default.WithLanguageVersion(LanguageVersion.Preview)).GetCompilationUnitRoot();
        var typeDeclarations = root.DescendantNodes().OfType<TypeDeclarationSyntax>().Count();
        var methodLike = root.DescendantNodes().OfType<MethodDeclarationSyntax>().Count()
            + root.DescendantNodes().OfType<ConstructorDeclarationSyntax>().Count();
        return new OriginalCoverageCount(typeDeclarations, methodLike);
    }

    private static BoundIdentityRecord CaptureBoundIdentity(BoundModule module)
    {
        var functions = module.Functions.Select(f => new BoundFunctionRecord(
            Name: f.Symbol.Name,
            ContainingType: f.ContainingTypeName,
            MemberKind: f.MemberKind.ToString(),
            SymbolId: f.SymbolId.ToString(),
            ReturnType: f.Symbol.ReturnType,
            Parameters: f.Symbol.Parameters.Select(BoundVariableRecord.From).ToList())).ToList();
        var types = module.SymbolsById.Values
            .OfType<TypeSymbol>()
            .Select(TypeSymbolRecord.From)
            .ToList();

        var calls = EnumerateBoundNodes(module)
            .Where(n => n is BoundCallExpression or BoundCallStatement)
            .Select(CaptureBoundCall)
            .ToList();
        var news = EnumerateBoundNodes(module)
            .OfType<BoundNewExpression>()
            .Select(n => new BoundNewRecord(Type: CaptureBoundType(n.Type), ResolvedType: n.ResolvedType is null ? null : TypeSymbolRecord.From(n.ResolvedType)))
            .ToList();
        var returns = EnumerateBoundNodes(module)
            .OfType<BoundReturnStatement>()
            .Select(r => new BoundReturnRecord(r.Expression is null ? null : CaptureBoundType(r.Expression.Type)))
            .ToList();

        return new BoundIdentityRecord(functions, types, calls, news, returns);
    }

    private static IEnumerable<BoundNode> EnumerateBoundNodes(BoundNode root)
    {
        var stack = new Stack<BoundNode>();
        stack.Push(root);
        while (stack.Count > 0)
        {
            var node = stack.Pop();
            yield return node;
            foreach (var child in node.ChildNodes.Reverse())
                stack.Push(child);
            switch (node)
            {
                case BoundCallExpression { Receiver: not null } call:
                    stack.Push(call.Receiver);
                    break;
                case BoundCallStatement { Receiver: not null } call:
                    stack.Push(call.Receiver);
                    break;
            }
        }
    }

    private static BoundCallRecord CaptureBoundCall(BoundNode node)
    {
        if (node is BoundCallExpression expr)
        {
            return new BoundCallRecord(
                Kind: nameof(BoundCallExpression),
                Target: expr.Target,
                ResolvedTypeName: expr.ResolvedTypeName,
                ResolvedMethodName: expr.ResolvedMethodName,
                ResolvedParameterTypes: expr.ResolvedParameterTypes,
                ResolvedSymbol: expr.ResolvedSymbol is null ? null : FunctionSymbolRecord.From(expr.ResolvedSymbol),
                ReturnType: CaptureBoundType(expr.Type),
                ArgumentTypes: expr.Arguments.Select(a => CaptureBoundType(a.Type)).ToList(),
                ArgumentParameterIndices: expr.ArgumentParameterIndices,
                SelectedOverloadMatches: CaptureOverloadMatches(expr.SelectedOverloadMatches));
        }
        var stmt = (BoundCallStatement)node;
        return new BoundCallRecord(
            Kind: nameof(BoundCallStatement),
            Target: stmt.Target,
            ResolvedTypeName: stmt.ResolvedTypeName,
            ResolvedMethodName: stmt.ResolvedMethodName,
            ResolvedParameterTypes: stmt.ResolvedParameterTypes,
            ResolvedSymbol: stmt.ResolvedSymbol is null ? null : FunctionSymbolRecord.From(stmt.ResolvedSymbol),
            ReturnType: null,
            ArgumentTypes: stmt.Arguments.Select(a => CaptureBoundType(a.Type)).ToList(),
            ArgumentParameterIndices: stmt.ArgumentParameterIndices,
            SelectedOverloadMatches: CaptureOverloadMatches(stmt.SelectedOverloadMatches));
    }

    private static List<Dictionary<string, object?>> CaptureOverloadMatches(IEnumerable<object> matches)
    {
        var result = new List<Dictionary<string, object?>>();
        foreach (var match in matches)
        {
            var dict = new Dictionary<string, object?>(StringComparer.Ordinal)
            {
                ["type"] = match.GetType().FullName,
                ["display"] = match.ToString(),
            };
            foreach (var prop in match.GetType().GetProperties(BindingFlags.Public | BindingFlags.Instance))
            {
                if (prop.GetIndexParameters().Length != 0) continue;
                try
                {
                    var value = prop.GetValue(match);
                    dict[prop.Name] = ToJsonSafe(value);
                }
                catch (Exception ex) when (ex is not OutOfMemoryException)
                {
                    dict[prop.Name] = "unavailable: " + ex.GetType().Name + ": " + ex.Message;
                }
            }
            result.Add(dict);
        }
        return result;
    }

    private static BoundTypeRecord CaptureBoundType(BoundType type)
    {
        var common = new BoundTypeRecord
        {
            Kind = type.GetType().Name,
            DisplayString = type.DisplayString,
            FullyQualifiedName = type.FullyQualifiedName,
        };
        if (type is NominalBoundType nominal)
        {
            common.QualifiedName = nominal.QualifiedName;
            common.NullableAnnotation = nominal.NullableAnnotation.ToString();
            common.Declaration = nominal.Declaration is null ? null : TypeSymbolRecord.From(nominal.Declaration);
            common.RoslynSymbol = nominal.RoslynSymbol is null ? null : RoslynSymbolRecord.From(nominal.RoslynSymbol);
            common.IsKnownReferenceType = ReadInternalBool(nominal, "IsKnownReferenceType");
        }
        return common;
    }

    private static BinderMetadataProfileRecord CaptureBinderMetadataProfile(Binder binder)
    {
        var field = typeof(Binder).GetField("_metadataBinder", BindingFlags.Instance | BindingFlags.NonPublic);
        var metadataBinder = field?.GetValue(binder);
        if (metadataBinder is null)
            return BinderMetadataProfileRecord.NotInstantiated();

        try
        {
            var contextProperty = metadataBinder.GetType().GetProperty("Context", BindingFlags.Instance | BindingFlags.NonPublic);
            var context = contextProperty?.GetValue(metadataBinder);
            if (context is null)
                return BinderMetadataProfileRecord.Unavailable("MetadataBinder.Context reflection returned null.");
            var hostProperty = context.GetType().GetProperty("HostCompilationForBinder", BindingFlags.Instance | BindingFlags.NonPublic);
            var compilation = hostProperty?.GetValue(context) as CSharpCompilation;
            if (compilation is null)
                return BinderMetadataProfileRecord.Unavailable("HostCompilationForBinder reflection returned null.");
            var refs = CaptureReferences(compilation.References);
            return new BinderMetadataProfileRecord("instantiated", null, refs, HashReferenceProfile(refs));
        }
        catch (Exception ex) when (ex is not OutOfMemoryException)
        {
            return BinderMetadataProfileRecord.Unavailable(ex.GetType().Name + ": " + ex.Message);
        }
    }

    private static ReferenceProfileRecord CaptureReferenceProfile()
    {
        var refs = CaptureReferences(GeneratedCSharpCompiler.References);
        return new ReferenceProfileRecord(refs, HashReferenceProfile(refs));
    }

    private static List<ReferenceRecord> CaptureReferences(IEnumerable<MetadataReference> references)
    {
        var records = new List<ReferenceRecord>();
        var index = 0;
        foreach (var reference in references)
        {
            string? path = (reference as PortableExecutableReference)?.FilePath;
            string? sha = null;
            string? hashError = null;
            string? identity = null;
            try
            {
                if (!string.IsNullOrEmpty(path) && File.Exists(path))
                {
                    sha = Sha256File(path);
                    identity = AssemblyName.GetAssemblyName(path).FullName;
                }
                else if (!string.IsNullOrEmpty(reference.Display))
                {
                    hashError = "reference has no existing FilePath; Display=" + reference.Display;
                }
                else
                {
                    hashError = "reference has no Display or existing FilePath";
                }
            }
            catch (Exception ex) when (ex is not OutOfMemoryException)
            {
                hashError = ex.GetType().Name + ": " + ex.Message;
            }
            records.Add(new ReferenceRecord(
                Ordinal: index++,
                Display: reference.Display,
                FilePath: path,
                Sha256: sha,
                HashError: hashError,
                AssemblyIdentity: identity,
                Aliases: reference.Properties.Aliases.IsDefault ? [] : reference.Properties.Aliases.ToArray(),
                EmbedInteropTypes: reference.Properties.EmbedInteropTypes));
        }
        return records;
    }

    private static SourceContextRecord CaptureSourceContext(DriverOptions options)
    {
        static SourceFileHash HashUnderRoot(string root, string relative)
        {
            var full = Path.GetFullPath(Path.Combine(root, relative));
            return File.Exists(full)
                ? new SourceFileHash(relative, Sha256File(full), null)
                : new SourceFileHash(relative, null, "missing");
        }
        var files = new[]
        {
            "Directory.Build.props",
            "Directory.Packages.props",
            "src/Calor.Compiler/Calor.Compiler.csproj",
            "src/Calor.Compiler/Migration/CSharpToCalorConverter.cs",
            "src/Calor.Compiler/Migration/RoslynSyntaxVisitor.cs",
            "src/Calor.Compiler/Migration/CalorEmitter.cs",
            "src/Calor.Compiler/Parsing/Parser.cs",
            "src/Calor.Compiler/Binding/Binder.cs",
            "src/Calor.Compiler/Binding/BoundTypes/BoundType.cs",
            "src/Calor.Compiler/CodeGen/CSharpEmitter.cs",
        }.Select(f => HashUnderRoot(options.CalorRoot, f)).ToList();
        return new SourceContextRecord(files);
    }

    private static void ValidateExpectedSourceHashes(EvidenceReport report, DriverOptions options)
    {
        foreach (var (relative, expected) in options.ExpectedFileHashes)
        {
            var full = Path.GetFullPath(Path.Combine(options.CalorRoot, relative));
            if (!File.Exists(full))
            {
                report.HardFailures.Add(new FailureRecord("source-pin-missing", relative, "Expected source file is missing."));
                continue;
            }
            var actual = Sha256File(full);
            if (!string.Equals(actual, expected, StringComparison.OrdinalIgnoreCase))
            {
                report.HardFailures.Add(new FailureRecord("source-pin-mismatch", relative, $"expected={expected}; actual={actual}"));
            }
        }
    }

    private static bool RuntimeShapeSatisfied(IReadOnlyDictionary<string, object?>? values)
    {
        if (values is null) return false;
        return ReadBool(values, "NullEchoIsNull") == true
            && ReadBool(values, "PresentEchoSameReference") == true
            && ReadBool(values, "DownstreamPresentSameReference") == true
            && ReadBool(values, "FreshIsNotNull") == true
            && ReadBool(values, "FreshIsDistinctFromPresent") == true
            && ReadBool(values, "NullDownstreamReadCompleted") == false
            && string.Equals(ReadString(values, "NullDownstreamReadExceptionType"), typeof(NullReferenceException).FullName, StringComparison.Ordinal);
    }

    private static IReadOnlyDictionary<string, object?> CapturePublicValues(object? value)
    {
        if (value is null) return new Dictionary<string, object?> { ["<returned>"] = null };
        var dict = new SortedDictionary<string, object?>(StringComparer.Ordinal);
        var type = value.GetType();
        foreach (var prop in type.GetProperties(BindingFlags.Public | BindingFlags.Instance))
        {
            if (prop.GetIndexParameters().Length == 0 && prop.GetMethod is not null)
                dict[prop.Name] = ToJsonSafe(prop.GetValue(value));
        }
        foreach (var field in type.GetFields(BindingFlags.Public | BindingFlags.Instance))
            dict[field.Name] = ToJsonSafe(field.GetValue(value));
        return dict;
    }

    private static object? ToJsonSafe(object? value)
    {
        if (value is null || value is string || value is bool || value is int || value is long || value is double || value is decimal)
            return value;
        if (value is Enum e) return e.ToString();
        if (value is ImmutableArray<int> iai && !iai.IsDefault) return iai.ToArray();
        if (value is IEnumerable enumerable && value is not string)
        {
            var items = new List<object?>();
            foreach (var item in enumerable)
                items.Add(ToJsonSafe(item));
            return items;
        }
        return value.ToString();
    }

    private static bool? ReadInternalBool(object instance, string propertyName)
    {
        try
        {
            return instance.GetType().GetProperty(propertyName, BindingFlags.Instance | BindingFlags.NonPublic | BindingFlags.Public)?.GetValue(instance) as bool?;
        }
        catch
        {
            return null;
        }
    }

    private static bool? ReadBool(IReadOnlyDictionary<string, object?> values, string name) =>
        values.TryGetValue(name, out var value) && value is bool b ? b : null;

    private static string? ReadString(IReadOnlyDictionary<string, object?> values, string name) =>
        values.TryGetValue(name, out var value) ? value?.ToString() : null;

    private static string StripNullablePrefix(string? typeName)
    {
        if (string.IsNullOrWhiteSpace(typeName)) return string.Empty;
        return typeName.StartsWith('?', StringComparison.Ordinal) ? typeName[1..] : typeName.EndsWith('?', StringComparison.Ordinal) ? typeName[..^1] : typeName;
    }

    private static bool IsRoslynError(RoslynDiagnostic d) => d.Severity == Microsoft.CodeAnalysis.DiagnosticSeverity.Error;

    private static List<DiagnosticRecord> Diagnostics(DiagnosticBag bag) => bag.Select(DiagnosticRecord.From).ToList();

    private static string Sha256Text(string text) => Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(text))).ToLowerInvariant();

    private static string Sha256File(string path) => Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(path))).ToLowerInvariant();

    private static string HashReferenceProfile(IReadOnlyList<ReferenceRecord> refs) => Sha256Text(string.Join("\n", refs.Select(r => $"{r.Ordinal}|{r.AssemblyIdentity}|{r.FilePath}|{r.Sha256}|{r.HashError}")));

    private static string SanitizeAssemblyName(string name) => new(name.Select(ch => char.IsLetterOrDigit(ch) ? ch : '_').ToArray());

    private static void WriteReport(EvidenceReport report, string? outputPath)
    {
        var json = JsonSerializer.Serialize(report, JsonOptions);
        if (string.IsNullOrWhiteSpace(outputPath))
        {
            Console.WriteLine(json);
            return;
        }
        Directory.CreateDirectory(Path.GetDirectoryName(Path.GetFullPath(outputPath)) ?? ".");
        File.WriteAllText(outputPath, json, Encoding.UTF8);
        Console.WriteLine(outputPath);
    }

    private static OverallRecord Summarize(EvidenceReport report)
    {
        var route = report.FaithfulCandidate?.Status == "passed-fixture-only"
            ? "faithful-fixture-specific-nullable-reference-representation"
            : report.RawInteropVariant?.Runtime?.RuntimeExpectationSatisfied == true
                ? "raw-interop-fallback-preserves-runtime-with-native-coverage-loss"
                : "deferred-or-failed";
        return new OverallRecord(route, report.HardFailures.Count == 0, report.HardFailures.Select(f => f.Id).ToList());
    }
}

internal sealed class FixtureNullableAstTransformer
{
    private readonly ISet<string> _methodNames;
    public List<ChangedSiteRecord> ChangedSites { get; } = [];

    public FixtureNullableAstTransformer(ISet<string> methodNames) => _methodNames = methodNames;

    public ModuleNode TransformModule(ModuleNode module)
    {
        var useSourceOrderedItems = module.Items.Count > 0;
        var classes = useSourceOrderedItems ? module.Classes : module.Classes.Select(TransformClass).ToList();
        var items = useSourceOrderedItems ? module.Items.Select(TransformModuleItem).ToList() : module.Items;
        var transformed = new ModuleNode(
            module.Span,
            module.Id,
            module.Name,
            module.Usings,
            module.Interfaces,
            classes,
            module.Enums,
            module.EnumExtensions,
            module.Delegates,
            module.Functions,
            module.Attributes,
            module.Issues,
            module.Assumptions,
            module.Invariants,
            module.Decisions,
            module.Context,
            module.InteropBlocks,
            module.RefinementTypes,
            module.IndexedTypes,
            module.TypePreprocessorBlocks,
            module.IdentifierSpan,
            items,
            module.NamespaceScopes,
            module.DeclaredSemanticsVersion);
        CopyMetadata(module, transformed);
        return transformed;
    }

    private AstNode TransformModuleItem(AstNode item) => item switch
    {
        ClassDefinitionNode cls => TransformClass(cls),
        _ => item,
    };

    private AstNode TransformClassItem(AstNode item) => item switch
    {
        MethodNode method => TransformMethod(method),
        ClassDefinitionNode cls => TransformClass(cls),
        _ => item,
    };

    private ClassDefinitionNode TransformClass(ClassDefinitionNode cls)
    {
        var useSourceOrderedItems = cls.Items.Count > 0;
        var methods = useSourceOrderedItems ? cls.Methods : cls.Methods.Select(TransformMethod).ToList();
        var nestedClasses = useSourceOrderedItems ? cls.NestedClasses : cls.NestedClasses.Select(TransformClass).ToList();
        var items = useSourceOrderedItems ? cls.Items.Select(TransformClassItem).ToList() : cls.Items;
        var transformed = new ClassDefinitionNode(
            cls.Span,
            cls.Id,
            cls.Name,
            cls.IsAbstract,
            cls.IsSealed,
            cls.IsPartial,
            cls.IsStatic,
            cls.BaseClass,
            cls.ImplementedInterfaces,
            cls.TypeParameters,
            cls.Fields,
            cls.Properties,
            cls.Constructors,
            methods,
            cls.Events,
            cls.OperatorOverloads,
            cls.Attributes,
            cls.CSharpAttributes,
            cls.IsStruct,
            cls.IsReadOnly,
            cls.Visibility,
            cls.InteropBlocks,
            cls.PreprocessorBlocks,
            nestedClasses,
            cls.NestedInterfaces,
            cls.NestedEnums,
            cls.Indexers,
            cls.NestedDelegates,
            cls.IdentifierSpan,
            cls.BaseClassSpan,
            cls.ImplementedInterfaceSpans,
            items);
        transformed.SourceFile = cls.SourceFile;
        CopyMetadata(cls, transformed);
        return transformed;
    }

    private MethodNode TransformMethod(MethodNode method)
    {
        if (!_methodNames.Contains(method.Name)) return method;
        var parameters = method.Parameters.Select(p => TransformParameter(method, p)).ToList();
        var output = method.Output is null ? null : TransformOutput(method, method.Output);
        var transformed = new MethodNode(
            method.Span,
            method.Id,
            method.Name,
            method.Visibility,
            method.Modifiers,
            method.TypeParameters,
            parameters,
            output,
            method.Effects,
            method.Preconditions,
            method.Postconditions,
            method.Body,
            method.Attributes,
            method.CSharpAttributes,
            method.IsAsync,
            method.IdentifierSpan)
        {
            EffectParameters = method.EffectParameters,
        };
        CopyMetadata(method, transformed);
        return transformed;
    }

    private ParameterNode TransformParameter(MethodNode owner, ParameterNode parameter)
    {
        if (parameter.TypeName != "Item") return parameter;
        var transformed = new ParameterNode(
            parameter.Span,
            parameter.Name,
            "?Item",
            parameter.Modifier,
            parameter.Attributes,
            parameter.CSharpAttributes,
            parameter.DefaultValue,
            parameter.InlineRefinement,
            parameter.IdentifierSpan,
            parameter.TypeNameSpan,
            parameter.Row);
        CopyMetadata(parameter, transformed);
        ChangedSites.Add(new ChangedSiteRecord(owner.Name, "parameter", parameter.Name, parameter.TypeName, transformed.TypeName));
        return transformed;
    }

    private OutputNode TransformOutput(MethodNode owner, OutputNode output)
    {
        if (output.TypeName != "Item") return output;
        var transformed = new OutputNode(output.Span, "?Item", output.TypeNameSpan, output.Row);
        CopyMetadata(output, transformed);
        ChangedSites.Add(new ChangedSiteRecord(owner.Name, "return", null, output.TypeName, transformed.TypeName));
        return transformed;
    }

    private static void CopyMetadata(AstNode source, AstNode target)
    {
        target.NamespaceIdentity = source.NamespaceIdentity;
        target.NamespaceScopeId = source.NamespaceScopeId;
        target.FullyQualifiedSymbolIdentity = source.FullyQualifiedSymbolIdentity;
        target.DocComment = source.DocComment;
    }
}

internal sealed record DriverOptions(string CalorRoot, string? OutputPath, string ConfigurationLabel, IReadOnlyDictionary<string, string> ExpectedFileHashes)
{
    public static DriverOptions Parse(string[] args)
    {
        string? calorRoot = null;
        string? output = null;
        var label = "current";
        var expected = new Dictionary<string, string>(StringComparer.Ordinal);
        for (var i = 0; i < args.Length; i++)
        {
            switch (args[i])
            {
                case "--calor-root":
                    calorRoot = RequireValue(args, ref i, "--calor-root");
                    break;
                case "--output":
                    output = RequireValue(args, ref i, "--output");
                    break;
                case "--configuration-label":
                    label = RequireValue(args, ref i, "--configuration-label");
                    break;
                case "--expect-file-sha":
                    var pin = RequireValue(args, ref i, "--expect-file-sha");
                    var equals = pin.IndexOf('=');
                    if (equals <= 0 || equals == pin.Length - 1)
                        throw new ArgumentException("--expect-file-sha must be relative/path=sha256");
                    expected[pin[..equals]] = pin[(equals + 1)..];
                    break;
                default:
                    throw new ArgumentException("Unknown argument: " + args[i]);
            }
        }
        if (string.IsNullOrWhiteSpace(calorRoot))
            throw new ArgumentException("--calor-root is required and must point at the scratch compiler source root.");
        var compilerProject = Path.Combine(calorRoot, "src", "Calor.Compiler", "Calor.Compiler.csproj");
        if (!File.Exists(compilerProject))
            throw new DirectoryNotFoundException("Calor compiler project not found under --calor-root: " + compilerProject);
        return new DriverOptions(calorRoot, output, label, expected);
    }

    public static string? TryGetOutputPath(string[] args)
    {
        for (var i = 0; i < args.Length - 1; i++)
            if (args[i] == "--output") return args[i + 1];
        return null;
    }

    private static string RequireValue(string[] args, ref int i, string name)
    {
        if (i + 1 >= args.Length) throw new ArgumentException(name + " requires a value.");
        i++;
        return args[i];
    }
}

internal sealed record RoslynSymbolRecord(string Kind, string Name, string MetadataName, string AssemblyIdentity, bool IsReferenceType, string NullableAnnotation)
{
    public static RoslynSymbolRecord From(ISymbol symbol)
    {
        var type = symbol switch
        {
            IMethodSymbol method => method.ReturnType,
            IParameterSymbol parameter => parameter.Type,
            ITypeSymbol ts => ts,
            _ => null,
        };
        return new RoslynSymbolRecord(
            symbol.Kind.ToString(),
            symbol.Name,
            symbol.ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat),
            symbol.ContainingAssembly?.Identity.GetDisplayName() ?? "unavailable",
            type?.IsReferenceType ?? false,
            type?.NullableAnnotation.ToString() ?? "unavailable");
    }
}
