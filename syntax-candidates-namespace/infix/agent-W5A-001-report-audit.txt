module Report
  fn FormatHeader pub stat (string:title) -> string
    effects
    return "== " + (title + " ==")
  fn FormatLine pub stat (string:name, int:value) -> string
    effects
    return name + (": " + value.ToString())
  fn FormatSummary pub stat (string:title, int:total) -> string
    effects
    let string:header = FormatHeader(title)
    let string:body = FormatLine("total", total)
    return header + (" | " + body)
  fn WriteReport pub stat (string:path, string:title, int:total) -> void
    effects fs:w
    let string:content = FormatSummary(title, total)
    File.WriteAllText(path, content)
  fn FormatFooter pub stat (string:generatedBy) -> string
    effects
    return "-- generated by " + (generatedBy + " --")
  fn WriteReportWithFooter pub stat (string:path, string:title, int:total, string:generatedBy) -> void
    effects fs:w
    let string:summary = FormatSummary(title, total)
    let string:footer = FormatFooter(generatedBy)
    let string:content = summary + (" | " + footer)
    File.WriteAllText(path, content)
