// Visual parity: screenshot the same pages from the old (next/font/google) and new
// (next/font/local) builds after fonts load, and compare the PNG bytes and the loaded faces.
import { createServer } from 'node:http';
import { readFile } from 'node:fs/promises';
import path from 'node:path';
import { createRequire } from 'node:module';
const require = createRequire('/Users/juanrivera/repos/juanmicrosoft/calor/.claude/worktrees/agent-a8d82faea8b9163ce/website/package.json');
const { chromium } = require('@playwright/test');

const S = '/private/tmp/claude-501/-Users-juanrivera-repos-juanmicrosoft-calor/98372634-e96e-40c2-8b28-3e5d5ca7761f/scratchpad';
const roots = { old: `${S}/old/website/out`, new: `${S}/newout` };
const types = { '.html': 'text/html', '.js': 'text/javascript', '.css': 'text/css', '.json': 'application/json',
  '.woff2': 'font/woff2', '.svg': 'image/svg+xml', '.png': 'image/png', '.webp': 'image/webp', '.jpg': 'image/jpeg', '.txt': 'text/plain' };
function serve(root, port) {
  return new Promise((ok) => {
    const server = createServer(async (req, res) => {
      const p = decodeURIComponent(new URL(req.url, 'http://x').pathname);
      const file = path.join(root, p.endsWith('/') ? `${p}index.html` : p);
      try { const body = await readFile(file); res.writeHead(200, { 'Content-Type': types[path.extname(file)] || 'application/octet-stream' }); res.end(body); }
      catch { res.writeHead(404).end(); }
    }).listen(port, '127.0.0.1', () => ok(server));
  });
}
const pages = ['/', '/docs/', '/docs/getting-started/', '/docs/changelog/', '/docs/benchmarking/', '/docs/syntax-reference/', '/docs/philosophy/', '/docs/cli/'];
const servers = [await serve(roots.old, 4181), await serve(roots.new, 4182)];
const browser = await chromium.launch();
const external = [];
let same = 0, differ = 0;
for (const theme of ['light', 'dark']) for (const width of [1366, 390]) for (const page of pages) {
  const shots = {};
  const faces = {};
  for (const [name, port] of [['old', 4181], ['new', 4182]]) {
    const ctx = await browser.newContext({ viewport: { width, height: 900 }, colorScheme: theme, reducedMotion: 'reduce' });
    const tab = await ctx.newPage();
    tab.on('request', (r) => { const u = new URL(r.url()); if (u.hostname !== '127.0.0.1') external.push(`${name} ${u.hostname}`); });
    await tab.route(/^https?:\/\/(?!127\.0\.0\.1)/, (route) => route.abort());
    const resp = await tab.goto(`http://127.0.0.1:${port}${page}`, { waitUntil: 'networkidle' });
    if (!resp || resp.status() !== 200) { console.log(`skip ${page} (${resp?.status()})`); await ctx.close(); continue; }
    await tab.evaluate(async () => { await document.fonts.ready; });
    await tab.addStyleTag({ content: 'video{visibility:hidden!important} *{animation:none!important;transition:none!important;caret-color:transparent!important}' });
    await tab.waitForTimeout(300);
    faces[name] = await tab.evaluate(() => [...document.fonts].filter((f) => f.status === 'loaded')
      .map((f) => `${f.family.replace(/_[0-9a-f]{6}$/, '').replace(/^'|'$/g, '')} ${f.weight}`).sort());
    shots[name] = await tab.screenshot({ fullPage: true });
    await ctx.close();
  }
  if (!shots.old || !shots.new) continue;
  const eq = Buffer.compare(shots.old, shots.new) === 0;
  eq ? same++ : differ++;
  console.log(`${eq ? 'SAME  ' : 'DIFFER'} ${theme} ${width} ${page} loaded faces old=${faces.old.length} new=${faces.new.length}`);
  if (!eq) {
    const fs = await import('node:fs');
    const tag = `${theme}-${width}-${page.replace(/\W+/g, '_')}`;
    fs.writeFileSync(`${S}/parity-${tag}-old.png`, shots.old);
    fs.writeFileSync(`${S}/parity-${tag}-new.png`, shots.new);
  }
}
console.log(`same=${same} differ=${differ}`);
console.log(`external requests (blocked): ${[...new Set(external)].join(', ') || 'none'}`);
await browser.close();
servers.forEach((s) => s.close());
