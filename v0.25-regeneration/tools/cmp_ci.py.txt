"""Compare local clone A build hashes with the Linux CI reproducible-builds hashes; print SDK fingerprints."""
import json, sys
C25 = "/private/tmp/claude-501/-Users-juanrivera-repos-juanmicrosoft-calor/98372634-e96e-40c2-8b28-3e5d5ca7761f/scratchpad/c25"
a = json.load(open(f"{C25}/out/clone-a/build-hashes.json"))
rb = f"{C25}/ci/rb-39323636839"
out = {"local": "clone-a (macOS arm64)", "ci": "reproducible-builds run 39323636839 (Linux x86_64, two builds)", "rows": [], "toolchains": {}}
ok = True
for n in ("packages-1", "packages-2"):
    c = json.load(open(f"{rb}/reproducible-packages-hashes/{n}.json"))
    out["toolchains"][n] = c.get("toolchain")
    for k, v in a["packages"].items():
        cv = c["packages"].get(k, {})
        same = cv.get("sha256") == v["sha256"] and cv.get("entries") == v["entries"]
        ok &= same
        out["rows"].append({"ci": n, "file": k, "local": v["sha256"], "ciSha256": cv.get("sha256"), "same": same})
    for k, v in a["releaseMetadata"].items():
        lv = v if isinstance(v, str) else v.get("sha256")
        cv = c["releaseMetadata"].get(k)
        cv = cv if isinstance(cv, str) or cv is None else cv.get("sha256")
        same = lv == cv; ok &= same
        out["rows"].append({"ci": n, "file": k, "local": lv, "ciSha256": cv, "same": same})
for n in ("website-1", "website-2"):
    c = json.load(open(f"{rb}/reproducible-website-hashes/{n}.json"))
    out["toolchains"][n] = c.get("toolchain")
    same = c["website"]["treeSha256"] == a["website"]["treeSha256"] and c["website"]["files"] == a["website"]["files"]
    ok &= same
    out["rows"].append({"ci": n, "file": f"website tree ({a['website']['fileCount']} files)", "local": a["website"]["treeSha256"], "ciSha256": c["website"]["treeSha256"], "same": same})
out["localToolchain"] = a["toolchain"]
out["allSame"] = ok
json.dump(out, open(sys.argv[1], "w"), indent=1)
print(json.dumps(out, indent=1))
