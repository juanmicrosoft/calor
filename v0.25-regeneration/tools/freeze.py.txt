#!/usr/bin/env python3
"""raw-artifact-freeze.json: SHA-256 of every re-proof file (this directory and the six family ready/ trees) plus the
archived files' index, in the role the R2 gate reads as an evidence manifest. usage: freeze.py <worktree>"""
import hashlib, json, os, sys
W = sys.argv[1]
REL = "docs/plans/evidence/v0.25-regeneration"
FAM = ["v0.25-f1-943", "v0.25-f2-906", "v0.25-f3-847", "v0.25-f4-1139", "v0.25-f5-1132", "v0.25-f6-1144"]
files = {}
for base in [REL] + [f"docs/plans/evidence/{f}/ready" for f in FAM]:
    for root, _, fs in os.walk(f"{W}/{base}"):
        for f in fs:
            p = os.path.relpath(os.path.join(root, f), W)
            if p == f"{REL}/raw-artifact-freeze.json":
                continue
            files[p] = hashlib.sha256(open(os.path.join(W, p), "rb").read()).hexdigest()
arch = json.load(open(f"{W}/{REL}/archives.json"))
pub = json.load(open(f"{W}/{REL}/publication-candidates.json"))["publication"]
out = {"schema": "v0.25-regeneration/raw-artifact-freeze/1", "role": "raw-artifact-freeze", "issue": 1430,
       "candidate": {"commit": "48cae911700ceb0dfc358e00933650445d81604f", "version": "0.25.0"},
       "candidateManifest": {"path": "docs/plans/evidence/v0.25-candidate/candidate-manifest.json", "landing": "51249109324aaff265202e73097721c505b89634"},
       "status": "0.25 re-proof on the candidate's own SHA: every check PASSED except provenance-determinism, BLOCKED (protocol 1.6.0 execution v025-reproof-1 INCOMPLETE). F2 is BLOCKED (F2-INTERP-09) and F4 is BLOCKED (F4-ITER-04); F1, F3, F5 and F6 have READY records pending the approver URL (A1).",
       "publication": pub, "files": dict(sorted(files.items())), "archivedFiles": arch["files"],
       "note": "Covers the files of the measurement commit. Later additions in this directory (r2-dry-run/, t1-aggregate.json, classifier/, regeneration.md and reviews/) are not in this list."}
json.dump(out, open(f"{W}/{REL}/raw-artifact-freeze.json", "w"), indent=1)
print(len(files), "files frozen")
