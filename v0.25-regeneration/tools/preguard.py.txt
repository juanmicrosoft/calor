import json, os, subprocess, sys
root = sys.argv[1]; execution_id = sys.argv[2]; commit = sys.argv[3]
sys.path.insert(0, os.path.join(root, "scripts"))
os.chdir(root)
import determinism_runner as dr, determinism_protocol as dp
from pathlib import Path
token = subprocess.run(["gh", "auth", "token"], capture_output=True, text=True).stdout.strip()
protocol = dp.load(Path(root), dp.PROTOCOL)
inv = dr.fetch_inventory("juanmicrosoft/calor", token)
fake = {"id": 0, "runNumber": max(r["runNumber"] for r in inv) + 1, "title": f"determinism execution {execution_id}",
        "event": "workflow_dispatch", "status": "in_progress", "conclusion": None, "headSha": commit, "minutes": 0, "attemptsStarted": False}
env = {"GITHUB_EVENT_NAME": "workflow_dispatch", "GITHUB_SHA": commit, "GITHUB_RUN_ID": "0", "GITHUB_REPOSITORY": "juanmicrosoft/calor"}
probs = dr.plan_problems(protocol, mode="execution", execution_id=execution_id, env=env, inventory=inv + [fake],
    ledger=dr.load_ledger(protocol), changed_paths=dr.changed_paths,
    pr_merge_commit=lambda pr: dr.pr_merge_commit("juanmicrosoft/calor", token, pr),
    manifest_candidate=lambda p: dr.manifest_candidate(p, "origin/main"))
print(json.dumps({"inventory": [{k: r[k] for k in ("id", "runNumber", "title", "status", "conclusion", "headSha", "minutes")} for r in inv],
                  "simulatedRunNumber": fake["runNumber"], "problems": probs, "worstCase": dp.worst_case(protocol)}, indent=1))
