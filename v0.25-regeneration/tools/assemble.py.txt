#!/usr/bin/env python3
"""Copy the re-proof outputs into docs/plans/evidence/v0.25-regeneration/ (data/text extensions only), moving
files over ~1 MB to a staging tree for the orphan archive branch. usage: assemble.py <worktree>"""
import gzip, json, os, shutil, sys, glob, hashlib

C25 = "/private/tmp/claude-501/-Users-juanrivera-repos-juanmicrosoft-calor/98372634-e96e-40c2-8b28-3e5d5ca7761f/scratchpad/c25"
W = sys.argv[1]
REL = "docs/plans/evidence/v0.25-regeneration"
DST = f"{W}/{REL}"
ARCH = f"{C25}/archive-stage"
LIMIT = 1_000_000
DATA = (".json", ".jsonl", ".md", ".txt", ".log", ".trx", ".csv", ".gz", ".sha256")
archived = {}


def put(src, rel, gz=False):
    """Copy src to REL/rel (gzip when asked); a file over LIMIT goes to the archive stage instead."""
    rel = rel + (".gz" if gz else "")
    assert rel.endswith(DATA), rel
    assert not any(seg.startswith(".") for seg in rel.split("/")), rel
    data = open(src, "rb").read()
    if gz:
        data = gzip.compress(data, mtime=0)
    target_rel = f"{REL}/{rel}"
    if len(data) > LIMIT:
        os.makedirs(os.path.dirname(f"{ARCH}/{target_rel}"), exist_ok=True)
        open(f"{ARCH}/{target_rel}", "wb").write(data)
        archived[target_rel] = {"sha256": hashlib.sha256(data).hexdigest(), "size": len(data)}
        return
    os.makedirs(os.path.dirname(f"{DST}/{rel}"), exist_ok=True)
    open(f"{DST}/{rel}", "wb").write(data)


def tree(src_dir, rel_dir, skip=()):
    for root, dirs, files in os.walk(src_dir):
        dirs[:] = [d for d in dirs if d not in skip]
        for f in files:
            src = os.path.join(root, f)
            r = os.path.relpath(src, src_dir)
            if any(part in skip for part in r.split(os.sep)):
                continue
            if f.endswith(".trx"):
                put(src, f"{rel_dir}/{r}", gz=True)
            elif f.endswith(DATA):
                put(src, f"{rel_dir}/{r}")
            elif f.endswith((".sh", ".py", ".mdx", ".calr", ".cs", ".patch")):
                put(src, f"{rel_dir}/{r}.txt")


shutil.rmtree(DST, ignore_errors=True); shutil.rmtree(ARCH, ignore_errors=True)
O = f"{C25}/out"
# Local publication and fresh-clone lanes.
for lane, name in (("clone-a", "clone-a"), ("clone-b", "clone-b"), ("pub-a", "pub-a"), ("pub-b", "pub-b")):
    tree(f"{O}/{lane}", f"local/{name}", skip=("build",))
for f in ("compare-a-b.md", "compare-pub-a-b.md", "compare-pub-a-ci-packages.md", "compare-pub-a-ci-website.md", "compare-local-ci.json",
          "determinism-second-execution-preguard.json", "ci-test-totals.json"):
    sub = "determinism" if f.startswith("determinism") else ("ci" if f.startswith("ci-") else "local")
    put(f"{O}/{f}", f"{sub}/{f.replace('compare-local-ci.json', 'compare-clone-a-ci-before-runtime-10.0.13.json')}")
tree(f"{O}/discarded-attempt-1-clone-a", "local/discarded-attempt-1-clone-a", skip=("build", "tests", "vrd", "b1-regen"))
# Measurement and release-quality lanes.
tree(f"{O}/measure", "measure", skip=("f4probe",))
tree(f"{O}/release-quality", "release-quality")
# CI.
tree(f"{C25}/stage/ci", "ci")
tree(f"{C25}/stage/determinism", "determinism")
shutil.copy(f"{C25}/ci/det-39323628059/result-v025-reproof-1-r1/result.json", f"{DST}/determinism/result.json")
put(f"{C25}/ci/rb-39323636839/reproducible-packages-hashes/packages-1.json", "ci/reproducible-builds-39323636839/packages-1.json")
put(f"{C25}/ci/rb-39323636839/reproducible-packages-hashes/packages-2.json", "ci/reproducible-builds-39323636839/packages-2.json")
put(f"{C25}/ci/rb-39323636839/reproducible-website-hashes/website-1.json", "ci/reproducible-builds-39323636839/website-1.json")
put(f"{C25}/ci/rb-39323636839/reproducible-website-hashes/website-2.json", "ci/reproducible-builds-39323636839/website-2.json")
put(f"{C25}/ci/t2-39323635472/tier2-verification/tier2-fixtures.json", "ci/tier2-fixtures-39323635472.json")
# Producers (as .txt).
for f in sorted(glob.glob(f"{C25}/tools/*")):
    b = os.path.basename(f)
    if b in ("dotnet-install.sh",):
        continue
    put(f, f"tools/{b}.txt")
json.dump({"schema": "v0.25-regeneration/archives/1", "branch": "evidence/v0.25-archives", "commit": None,
           "rule": "Maintainer rule 2026-10-19 (as 0.24's 2026-10-08 decision): evidence files over about 1 MB live on the orphan branch evidence/v0.25-archives (never merged), at the same relative paths; this index replaces them here.",
           "verify": "git fetch origin evidence/v0.25-archives && git show <commit>:<path> | shasum -a 256",
           "files": dict(sorted(archived.items()))}, open(f"{DST}/archives.json", "w"), indent=1)
print("archived", len(archived), sum(v["size"] for v in archived.values()))
for k, v in sorted(archived.items()):
    print("  ", v["size"], k)
total = sum(os.path.getsize(os.path.join(r, f)) for r, _, fs in os.walk(DST) for f in fs)
print("in tree", sum(len(fs) for _, _, fs in os.walk(DST)), "files", total, "bytes")
