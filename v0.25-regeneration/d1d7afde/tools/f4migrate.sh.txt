#!/usr/bin/env bash
# F4-ITER-04: project-level `calor migrate` of the two FluentValidation projects that hold the three modules,
# on a copy of the pinned submodule (the clone's submodule is not modified).
C25=/private/tmp/claude-501/-Users-juanrivera-repos-juanmicrosoft-calor/98372634-e96e-40c2-8b28-3e5d5ca7761f/scratchpad/c26
USE_BREW_DOTNET=1 source $C25/tools/env.sh m
CAL=$C25/m/src/Calor.Compiler/bin/Release/net10.0/calor.dll
T=$C25/out/measure/f4-iter-04; mkdir -p $T
W=$C25/f4work; rm -rf $W; mkdir -p $W
cp -R $C25/m/bench/corpus/FluentValidation $W/FluentValidation
rm -rf $W/FluentValidation/.git
for p in FluentValidation.Tests FluentValidation; do
  /usr/bin/time -p dotnet $CAL migrate $W/FluentValidation/src/$p --skip-verify --skip-analyze --no-telemetry --report $T/migrate-$p.json > $T/migrate-$p.log 2>&1
  echo "$p exit $?"
done
for f in src/FluentValidation.Tests/EmptyTester src/FluentValidation.Tests/NotEmptyTester src/FluentValidation/TestHelper/ITestValidationContinuation; do
  if [ -f "$W/FluentValidation/$f.calr" ]; then cp "$W/FluentValidation/$f.calr" "$T/$(basename $f).migrate.calr"; echo "converted $f"; else echo "no output $f"; fi
done
