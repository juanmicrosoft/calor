#!/usr/bin/env bash
# 0.25 re-proof release-quality lane: every step of publish-nuget.yml's release-quality job, in order,
# without the workflow; then Tier 2, the Z3 release-binary mirror and the submodule pins.
# usage: rq.sh <tree> <out>
set -uo pipefail
TOOLS="$(cd "$(dirname "$0")" && pwd)"
source "$TOOLS/env.sh" rq
TREE="$1"; OUT="$2"
mkdir -p "$OUT"
cd "$TREE" || exit 2
CAND=d1d7afde672691613f7bb34d253e63401ee423ac
[ "$(git rev-parse HEAD)" = "$CAND" ] || { echo "HEAD is not the candidate"; exit 2; }
echo "start $(date -u +%FT%TZ) tree $TREE head $(git rev-parse HEAD) dotnet $(dotnet --version) runtimes: $(dotnet --list-runtimes | tr '\n' ';')" > "$OUT/steps.txt"
step() { local name="$1"; shift; "$@" > "$OUT/$name.log" 2>&1; local rc=$?; echo "STEP $name exit $rc at $(date -u +%FT%TZ)" >> "$OUT/steps.txt"; return $rc; }

step submodules git submodule update --init --recursive
git submodule status > "$OUT/submodule-status.txt"
step fetch-main-and-tags git fetch --no-recurse-submodules --force --tags origin +refs/heads/main:refs/remotes/origin/main
step download-z3 bash src/Calor.Compiler/scripts/download-z3.sh
build() { dotnet restore --locked-mode && dotnet build -c Release --no-restore && dotnet build tests/Calor.Performance.Tests/Calor.Performance.Tests.csproj -c Release; }
step restore-and-build build
selftests() {
  python3 scripts/check_test_quality.py --self-test && python3 scripts/check_trx.py --self-test && bash scripts/check_sample_ids.sh --self-test \
  && python3 scripts/check_coverage.py --self-test && python3 scripts/run_mutation_gate.py --self-test && python3 scripts/run_performance_gate.py --self-test \
  && python3 scripts/run_flake_gate.py --self-test && bash scripts/compiler-shard-filter.sh --self-test; }
step self-tests selftests
quality() { python3 scripts/check_test_quality.py && bash scripts/check_sample_ids.sh; }
step test-quality quality
coverage() {
  # Homebrew's dotnet host (0.24 C2 regeneration 3 practice): the private Microsoft root's path contains
  # "calor", which SoundnessRegistrationTests.EveryCaseOracleCompilesAgainstBclOnly rejects. Same Release binaries (--no-build).
  local DN=/opt/homebrew/bin/dotnet
  for project in Calor.Compiler.Tests Calor.Conversion.Tests Calor.Enforcement.Tests Calor.Semantics.Tests Calor.Verification.Tests; do
    if [ "$project" = "Calor.Compiler.Tests" ]; then
      for part in 1 2; do
        filter="$(bash scripts/compiler-shard-filter.sh --part "$part")"
        DOTNET_ROOT=/opt/homebrew/Cellar/dotnet/10.0.401_1/libexec $DN test "tests/$project/$project.csproj" -c Release --no-build --no-restore --filter "$filter" \
          --collect:"XPlat Code Coverage" --logger "trx;LogFileName=$project-$part.trx" --results-directory "artifacts/coverage/raw/$project" --verbosity quiet || return 1
      done
    else
      DOTNET_ROOT=/opt/homebrew/Cellar/dotnet/10.0.401_1/libexec $DN test "tests/$project/$project.csproj" -c Release --no-build --no-restore \
        --collect:"XPlat Code Coverage" --logger "trx;LogFileName=$project.trx" --results-directory "artifacts/coverage/raw/$project" --verbosity quiet || return 1
    fi
  done
  python3 scripts/check_coverage.py
}
step coverage coverage
step mutation python3 scripts/run_mutation_gate.py
step performance python3 scripts/run_performance_gate.py
step flake python3 scripts/run_flake_gate.py --runs 100
step roundtrip dotnet run --project tools/Calor.RoundTrip.Harness/ -c Release -- run --all --dotnet dotnet --output artifacts/migration/roundtrip
step roundtrip-delta python3 scripts/roundtrip_delta.py --baseline eng/roundtrip-baselines.json --reports artifacts/migration/roundtrip --output-markdown "$OUT/roundtrip-delta.md"
mkdir -p artifacts/migration
step scorecard dotnet run --project tests/Calor.Evaluation -c Release -- scorecard --format both -o artifacts/migration/conversion-scorecard
mkdir -p "$OUT/tier2"
t2() { python3 scripts/verify_corpus.py --report "$OUT/tier2/tier2-fixtures.json"; }
step tier2 t2
cp "$OUT/tier2.log" "$OUT/tier2/tier2.log"
z3rel() { python3 "$TOOLS/z3rel.py" .github/z3-binaries-4.15.7.sha256 "$OUT/z3-release-binaries.json" "$OUT/z3-release-download"; }
step z3-release z3rel
rm -rf "$OUT/z3-release-download"
tar czf "$OUT/artifacts.tar.gz" artifacts/coverage artifacts/mutation artifacts/performance artifacts/flake artifacts/migration 2>/dev/null
git status --porcelain > "$OUT/final-git-status.txt"
echo "end $(date -u +%FT%TZ)" >> "$OUT/steps.txt"
