def sha_text(b: bytes) -> str:
    return hashlib.sha256(b.decode("utf-8").replace("\r\n", "\n").encode("utf-8")).hexdigest()


def requirement_digest(c):
    obs = json.dumps(c.get("observations") if c.get("observations") is not None else [], separators=(",", ":"), ensure_ascii=False)
    # Sha256 = SHA-256 of the UTF-8 text after CRLF -> LF; cells/families joined with \n, empty when absent.
    s = (c.get("fixture") or "") + "\n" + (c.get("note") or "") + "\n" + "\n".join(c.get("cells") or []) + "\n" + obs \
        + "\n" + "\n".join(c.get("families") or []) + "\n" + (c.get("websiteExample") or "")
    return sha_text(s.encode("utf-8"))

def tests(lane, *patterns):
    """Every test in a lane's TRX whose name contains one of the patterns; fails closed on none."""
    out = []
    for pat in patterns:
        hits = [(n, o) for n, o in trx.get(lane, {}).items() if pat in n]
        if not hits:
            raise SystemExit(f"no test matching {pat!r} in {lane}")
        out.append({"trx": f"{REGEN}/measure/tests/{lane}.trx.gz", "match": pat, "count": len(hits),
                    "passed": sum(o == "Passed" for _, o in hits), "outcomes": sorted({o for _, o in hits})})
    return out


def tests_ok(ts):

def derive(s, original):
    """InteropScopeValidator.Derive, ported, plus the lossy/drop and refusal reading."""
    if not s:
        return None
    if (s.get("lossySubstitutionCount") or 0) > 0 or (s.get("dropCount") or 0) > 0:
        return "lossy"
    e = s.get("execution")
    if isinstance(e, dict):
        interop = any((s.get(k) or 0) > 0 for k in ("interopPreservationCount", "interopPreservations", "csharpInteropBlocks", "inlineInteropExpressions", "rawCSharpExpressions"))
        if "matchesOriginal" in e:
            match = e.get("compiled") is True and e.get("matchesOriginal") is True
        else:
            match = e.get("compiled") is True and (original or {}).get("compiled") is True and e.get("result") == (original or {}).get("result") \
                and e.get("exception") == (original or {}).get("exception")
        mode = e.get("compileMode") or s.get("generatedMode")
        return ("preserved" if interop else "native") + ("-match" if match else "-mismatch") + ("@permissive-effects" if mode == "permissive-effects" else "")
    if s.get("convertExit") not in (None, 0) or s.get("isError") is True or s.get("success") is False or s.get("migrateExit") not in (None, 0):
        return "rejected"
    k = s.get("compile")
    if isinstance(k, dict) and k.get("exit") not in (0, None):
        return "compile-error:" + ",".join(k.get("errorCodes") or [])
    return "no-output"


def category(outcome):
    if outcome is None:
        return None
    if outcome.endswith("-mismatch") or "-mismatch@" in outcome or outcome == "lossy":
        return "mismatch-or-lossy"
    if outcome.startswith("native-match"):
        return "native"
    if outcome.startswith("preserved-match"):
        return "preserved"
    if outcome == "rejected":
        return "rejected"
    return "invalid-output"

def driver_measure(src, case_id, surfaces=None):
    c = (R0 if src == "R0" else FD[src])[case_id]
    surf = {}
    for name, s in c["surfaces"].items():
        if surfaces and name not in surfaces:
            continue
        o = s.get("outcome") if src == "F1" else None
        d = derive(s, c.get("original"))
        surf[name] = {"outcome": d, "category": category(d), "driverOutcome": o}
    m = {"kind": "driver", "driver": DRIVER_PATH[src], "output": DRIVER_OUT[src], "case": case_id, "surfaces": surf}
    if src != "R0":
        diffs = [x for x in cmp[src]["behaviorDiffs"] if x.get("case") == case_id]
        same_refusal = lambda x: x.get("surface") == "cli-no-fallback" and all(x["committed"].get(k) == x["candidate"].get(k) for k in ("convertExit", "success", "errorCodes"))
        reporting = [x for x in diffs if same_refusal(x)]
        diffs = [x for x in diffs if not same_refusal(x)]
        m["vsApprovedEvidence"] = "same behavior on every surface as the family's approved evidence" if not diffs else diffs
        if reporting:
            m["reportingOnlyDifferences"] = {"surfaces": [x["surface"] for x in reporting], "note": "the --no-fallback refusal is unchanged (exit, success, error codes); on the candidate its JSON envelope also carries the counts and loss features (F6's reporting, which landed after the F4 branch measured)", "detail": reporting}
    return m
