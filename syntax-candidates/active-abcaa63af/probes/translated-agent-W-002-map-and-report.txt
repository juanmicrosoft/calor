§M{MapAfter}
  §F{Map:pub}<eff e> ([i32]:xs, Func<i32,i32>:f §E{e}) -> [i32]
    §E{e, alloc, mut}
    §B{n:i32} (len xs)
    §B{~out:[i32]} §ARR{i32} n
    §L{i:0:(- n 1):1}
      §SETIDX{out} i §C{f} §A §IDX xs i §/C
    §R out

  §F{Double:pub} (i32:x) -> i32
    §E{}
    §R (* x INT:2)

  §F{Announce:pub} (i32:x) -> i32
    §E{cw}
    §P x
    §R x

  §F{UsePure:pub} ([i32]:xs) -> [i32]

    §R §C{Map} §A xs §A Double §/C

  §F{UseImpure:pub} ([i32]:xs) -> [i32]

  §F{DoubleAndReport} (i32:x) -> i32
    §E{cw}
    §B{d:i32} (* x INT:2)
    §P d
    §R d

  §F{MapAndReport:pub} ([i32]:xs) -> [i32]
    §E{alloc, mut, cw}
    §R §C{Map} §A xs §A DoubleAndReport §/C

  §F{Total:pub} ([i32]:xs) -> i32
    §E{}
    §B{n:i32} (len xs)
    §B{~sum:i32} INT:0
    §L{i:0:(- n 1):1}
      §B{v:i32} §IDX xs i
      §ASSIGN sum (+ sum (* v INT:2))
    §R sum
