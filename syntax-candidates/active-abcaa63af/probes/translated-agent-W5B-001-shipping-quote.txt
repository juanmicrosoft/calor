§M{Quote}
  §F{BaseFee:pub} (i32:weight) -> i32
    §E{}
    §R (* weight 3)

  §F{DistanceFee:pub} (i32:distance) -> i32
    §E{}
    §R (* distance 2)

  §F{QuoteWithSurcharge:pub} (i32:baseAmount, i32:surcharge, i32:cap) -> i32
    §E{}
    §S (<= result cap)
    §B{total:i32} (+ baseAmount surcharge)
    §IF (> total cap)
      §R cap
    §R total

  §F{QuoteTotal:pub} (i32:weight, i32:distance, i32:cap) -> i32
    §E{}
    §B{baseAmount:i32} §C{BaseFee} §A weight §/C
    §B{extra:i32} §C{DistanceFee} §A distance §/C
    §R §C{QuoteWithSurcharge} §A baseAmount §A extra §A cap §/C

  §F{QuoteWithFloor:pub} (i32:baseAmount, i32:surcharge, i32:cap, i32:floor) -> i32
    §E{}
    §B{capped:i32} §C{QuoteWithSurcharge} §A baseAmount §A surcharge §A cap §/C
    §IF (< capped floor)
      §R floor
    §R capped

  §F{FormatQuote:pub} (i32:amount) -> str
    §E{}
    §R (concat "quote: " (str amount))
