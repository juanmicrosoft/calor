§M{Quota}
  §F{BaseAllowance:pub} (i32:tier) -> i32
    §E{}
    §R (* tier 5)

  §F{BurstBonus:pub} (i32:priority) -> i32
    §E{}
    §R (* priority 2)

  §F{GrantRequests:pub} (i32:requested, i32:maxAllowed) -> i32
    §E{}
    §S (<= result maxAllowed)
    §IF (> requested maxAllowed)
      §R maxAllowed
    §R requested

  §F{GrantForTier:pub} (i32:tier, i32:priority, i32:maxAllowed) -> i32
    §E{}
    §B{allowance:i32} §C{BaseAllowance} §A tier §/C
    §B{bonus:i32} §C{BurstBonus} §A priority §/C
    §B{requested:i32} (+ allowance bonus)
    §R §C{GrantRequests} §A requested §A maxAllowed §/C

  §F{GrantWithMinimum:pub} (i32:requested, i32:maxAllowed, i32:minGrant) -> i32
    §E{}
    §B{capped:i32} §C{GrantRequests} §A requested §A maxAllowed §/C
    §IF (< capped minGrant)
      §R minGrant
    §R capped

  §F{FormatGrant:pub} (i32:amount) -> str
    §E{}
    §R (concat "granted: " (str amount))
