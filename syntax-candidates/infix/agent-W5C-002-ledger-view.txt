module Ledger
  fn EntryLine pub stat (string:label, int:amount) -> string
    effects
    return label + ("=" + amount.ToString())
  fn AppendEntry pub stat (string:path, string:label, int:amount) -> void
    effects fs:w
    let string:line = EntryLine(label, amount)
    let string:record = line + ";"
    File.AppendAllText(path, record)
  fn ReadLedger pub stat (string:path) -> string
    effects fs:r
    let bool:exists = File.Exists(path)
    if !exists
      return ""
    return File.ReadAllText(path)
  fn ReportOf pub stat (string:content) -> string
    effects
    return "ledger[" + (content.Length.ToString() + "]")
  fn BalanceReport pub stat (string:path) -> string
    effects fs:r
    let string:content = ReadLedger(path)
    return ReportOf(content)
  fn HasEntries pub stat (string:path) -> bool
    effects fs:r
    let string:content = ReadLedger(path)
    return content.Length > 0
  fn DescribeLedger pub stat (string:path) -> string
    effects fs:r
    return "ledger: " + BalanceReport(path)
  fn LedgerSize pub stat (string:path) -> int
    effects fs:r
    let string:content = ReadLedger(path)
    return content.Length
