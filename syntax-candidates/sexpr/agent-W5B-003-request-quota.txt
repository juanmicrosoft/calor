  fn GrantForTier pub stat (int:tier, int:priority, int:maxAllowed) -> int
    effects
    let int:allowance = (BaseAllowance tier)
    let int:bonus = (BurstBonus priority)
    let int:requested = (+ allowance bonus)

  fn GrantWithMinimum pub stat (int:requested, int:maxAllowed, int:minGrant) -> int
    effects
    let int:capped = (GrantRequests requested maxAllowed)
    if (< capped minGrant)
      return minGrant
    return capped

    effects
