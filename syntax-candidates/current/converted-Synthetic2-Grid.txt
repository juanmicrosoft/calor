§M{m001:GeoLib:overflow=unchecked}
  §NS{ns001:GeoLib:file}
    // Simple integer-grid helpers — deliberately arithmetic/boundary-dense so the round-trip converter produces native regions and standard mutation operators (boundary, off-by-one, arithmetic, swap-return) have well-covered sites. A second corpus subject for Slice C.
    §CL{c002:Grid:pub:stat}
      §MT{m003:SumOfSquares:pub:stat} (i32:a, i32:b) -> i32
        §R (+ (* a a) (* b b))

      §MT{m004:Triple:pub:stat} (i32:x) -> i32
        §R (* x 3)

      §MT{m005:Area:pub:stat} (i32:width, i32:height) -> i32
        §R (* width height)

      §MT{m006:Perimeter:pub:stat} (i32:width, i32:height) -> i32
        §R (* 2 (+ width height))

      §MT{m007:InBounds:pub:stat} (i32:x, i32:y, i32:width, i32:height) -> bool
        §IF{if008} (|| (< x 0) (< y 0))
          §R false
        §IF{if009} (|| (>= x width) (>= y height))
          §R false
        §R true

      §MT{m010:ClampIndex:pub:stat} (i32:index, i32:length) -> i32
        §IF{if011} (< index 0)
          §R 0
        §IF{if012} (>= index length)
          §R (- length 1)
        §R index

      §MT{m013:ManhattanDistance:pub:stat} (i32:x1, i32:y1, i32:x2, i32:y2) -> i32
        §B{i32:dx} (- x1 x2)
        §B{i32:dy} (- y1 y2)
        §R (+ §C{Math.Abs} §A dx §/C §C{Math.Abs} §A dy §/C)

      §MT{m014:CellCount:pub:stat} (i32:rows, i32:cols) -> i32
        §B{~total:i32} 0
        §IF{scope026} true
          §B{~r:i32} 0
          §B{~_forFirst015:bool} true
          §WH{while025} true
            §IF{if016} _forFirst015
              §ASSIGN _forFirst015 false
            §EL
              §ASSIGN r (+ r 1)
            §IF{if017} (< r rows)
            §EL
              §BK
            §IF{body024} true
              §IF{scope023} true
                §B{~c:i32} 0
                §B{~_forFirst018:bool} true
                §WH{while022} true
                  §IF{if019} _forFirst018
                    §ASSIGN _forFirst018 false
                  §EL
                    §ASSIGN c (+ c 1)
                  §IF{if020} (< c cols)
                  §EL
                    §BK
                  §IF{body021} true
                    §ASSIGN total (+ total 1)
        §R total

      §MT{m027:IsSquare:pub:stat} (i32:width, i32:height) -> bool
        §R (== width height)
