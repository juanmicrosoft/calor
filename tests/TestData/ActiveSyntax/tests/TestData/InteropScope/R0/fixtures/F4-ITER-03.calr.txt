§M{ConvertedModule:overflow=unchecked}
  §U{System}
  §U{System.Collections.Generic}

  §NS{_global:global}
    §CL{Bag:pub}
      §FLD{i32:started:pub:stat} = 0

      §PROP{Items:Seq<i32>:pub}
        §GET{pub}
          §ASSIGN started (+ started 1)
          §YIELD 1
          §YIELD 2

      §IXER{Seq<i32>:pub}
        §GET{pub}
          §L{i:0:(- 10 1):1}
            §IF (>= i limit)
              §YBRK
            §YIELD i


    §M{Probe:pub}
      §F{Run:pub} () -> str
        §E{alloc}
        §B{bag} §NEW{Bag} §/NEW
        §B{items} bag.Items
        §B{i32:before} Bag.started
        §B{~sum:i32} 0
        §EACH{i:var} items
          §ASSIGN sum (+ sum i)
        §B{~count:i32} 0
        §EACH{i:var} §IDX{bag} 3
          §ASSIGN count (+ count 1)
        §R (+ (+ (+ (+ (+ (+ before "|") Bag.started) "|") sum) "|") count)
