Usage: generate_terminal_record.py <repo> <evidence commit> <out file>

Every evidence hash is SHA-256 over the exact blob bytes at <evidence commit> (the C2 regeneration-3
merge 12200dd7cf84cffd33a7bbe133b65b3cd03c3c2e on main). A directory pointer carries `treeSha256`:
SHA-256 over the sorted lines "<blob sha256>  <path relative to the directory>\\n" of every file under
it at that commit (R2's tree-digest convention). Nothing is read from the working tree.

Prepared by the A1 agent for the adjudicator (@juanmicrosoft). The agent proposes; the maintainer
decides by merging. The committed copy of this script is generate_terminal_record.py.txt.
"""

def git(*args: str) -> bytes:
    return subprocess.run(["git", "-C", REPO, *args], check=True, capture_output=True).stdout


def blob(path: str) -> bytes:
    return git("show", f"{EVIDENCE_COMMIT}:{path}")


def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def load(path: str) -> dict:
    return json.loads(blob(path))


def f(path: str) -> dict:
    return {"path": path, "sha256": sha(blob(path))}


def d(path: str) -> dict:
    path = path.rstrip("/") + "/"
    names = [n for n in git("ls-tree", "-r", "--name-only", EVIDENCE_COMMIT, "--", path).decode().splitlines() if n]
    if not names:
        raise SystemExit(f"empty or missing directory {path}")
    lines = "".join(f"{sha(blob(n))}  {n[len(path):]}\n" for n in sorted(names))
    return {"path": path, "files": len(names), "treeSha256": sha(lines.encode())}


def ev(*items: str) -> list[dict]:
    return [d(i) if i.endswith("/") else f(i) for i in items]

HISTORICAL_NOTE = ("Historical-only by the inventory (amendment 1.4.0 where reclassified). Withdrawn from 0.24 evidence: it establishes no 0.24 claim, and "
                   "this outcome does not make its numbers valid (contract §3).")


def artifact_rows() -> list[dict]:
    rows = []
    for a in inventory["artifacts"]:
        aid, cls = a["id"], a["classification"]
        spec = ARTIFACTS.get(aid, {})
        if cls == "stale":
            raise SystemExit(f"{aid} is still stale; it could only be BLOCKED")
        if cls != "historical-only" and not spec:
            raise SystemExit(f"no adjudication written for {aid}")
        resolved = {r["defect"]: f"Resolved by #{r['resolvedByPr']} ({r.get('scope', 'recorded in the inventory')})."
                    for r in a.get("defectResolutions", [])}
        resolved.update(spec.get("resolutions", {}))
        defects = []
        for text in a.get("openDefects", []):
            if cls == "historical-only":
                defects.append({"defect": text, "adjudication": "Not adjudicated as evidence: the artifact is historical-only."})
            elif text not in resolved:
                raise SystemExit(f"{aid}: cutoff defect without an adjudicated resolution: {text}")
            else:
                defects.append({"defect": text, "adjudication": resolved[text]})
        row = {
            "subject": aid,
            "kind": "artifact",
            "classification": cls,
            "outcome": "HISTORICAL-ONLY" if cls == "historical-only" else "BOUNDED",
            "independence": ROW,
            "limitation": limitation,
            "c2Status": c2[aid]["status"],
            "rowLimitations": spec.get("bounds", [HISTORICAL_NOTE] if cls == "historical-only" else []),
            "evidence": ev(*spec.get("evidence", [R3 + "results.json"])),
        }
        if cls == "historical-only" and "bounds" in spec:
            row["rowLimitations"] = [HISTORICAL_NOTE, *spec["bounds"]]
        if defects:
            row["cutoffDefects"] = defects
        rows.append(row)
    return rows

def gate_rows() -> list[dict]:
    manifest = load(EV + "c1-1423/candidate-manifest.json")
    merged = {c["issue"]: {p["pr"]: p["mergeCommit"] for p in c["mergedPrs"]} for c in manifest["acceptedChildren"]}
    extra = {1423: {1508: "fd34f41f0d5600b92b5868392e1ea3e1d1c197ff", 1534: "7eca486822ce81871df0cfb0a6e0ae16cf5ce7b9", 1539: C1_LANDING},
             1424: {1517: "d2288bf934dcda609e8719856dc6ed9e0a9cca3b", 1542: C2_MERGE}}
    for issue, prmap in extra.items():
        for commit in prmap.values():
            if git("rev-parse", commit).decode().strip() != commit or subprocess.run(
                    ["git", "-C", REPO, "merge-base", "--is-ancestor", commit, EVIDENCE_COMMIT]).returncode != 0:
                raise SystemExit(f"{commit} is not an ancestor of the evidence commit")
    rows = []
    for child in contract["children"]:
        issue = child["issue"]
        if issue == 1408:
            continue
        spec = GATES[issue]
        prs = []
        for pr in spec["prs"]:
            commit = merged.get(issue, {}).get(pr) or extra.get(issue, {}).get(pr)
            prs.append({"pr": pr, "mergeCommit": commit} if commit else {"pr": pr})
        row_extra = {}
        if issue in BLOCKED_GATES:
            row_extra["blockedBy"] = ("Contract §8 stopping rule 1: a §9 ceiling (review-rounds-per-pr, 3) was exceeded and no amendment "
                                      "raised it, so the gate is recorded BLOCKED. " + BLOCKED_GATES[issue])
        rows.append({
            "subject": f"gate:#{issue}",
            "kind": "gate",
            "gate": child.get("gate"),
            "outcome": "BLOCKED" if issue in BLOCKED_GATES else "BOUNDED",
            **row_extra,
            "independence": ROW,
            "limitation": limitation,
            "closureEvidenceRequired": child.get("closureEvidence"),
            "mergedPrs": prs,
            "rowLimitations": spec["bounds"],
            "evidence": ev(*spec["evidence"]),
        })
    return rows
